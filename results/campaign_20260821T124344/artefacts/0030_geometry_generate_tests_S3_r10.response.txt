Here is the complete C++ unit test suite covering the `Canvas` class.

```cpp
#include <iostream>
#include <cassert>
#include <cmath>
#include "Canvas.h"
#include "Shape.h"

// Helper function for floating point comparisons
bool areAlmostEqual(double a, double b, double epsilon = 1e-6) {
    return std::fabs(a - b) < epsilon;
}

// Test initial/empty state of Canvas
void test_Canvas_initialState() {
    Canvas canvas;
    assert(canvas.count() == 0);
    assert(canvas.shapes.empty());
    assert(areAlmostEqual(canvas.totalArea(), 0.0));
    assert(areAlmostEqual(canvas.largestArea(), 0.0));

    std::cout << "Test test_Canvas_initialState: PASS" << std::endl;
}

// Test adding null pointer
void test_Canvas_addNull() {
    Canvas canvas;
    canvas.add(nullptr);
    assert(canvas.count() == 0);
    assert(canvas.shapes.empty());

    std::cout << "Test test_Canvas_addNull: PASS" << std::endl;
}

// Test adding valid shapes and verifying count
void test_Canvas_addShapesAndCount() {
    Canvas canvas;
    canvas.add(new Rectangle(2.0, 3.0));
    assert(canvas.count() == 1);

    canvas.add(new Circle(1.0));
    assert(canvas.count() == 2);

    canvas.add(new Shape());
    assert(canvas.count() == 3);

    // Canvas destructor will clean up dynamically allocated shapes
    std::cout << "Test test_Canvas_addShapesAndCount: PASS" << std::endl;
}

// Test totalArea calculation
void test_Canvas_totalArea() {
    Canvas canvas;
    
    // Rectangle area: 4.0 * 5.0 = 20.0
    canvas.add(new Rectangle(4.0, 5.0));
    assert(areAlmostEqual(canvas.totalArea(), 20.0));

    // Circle area: PI * 2^2 = 4 * PI
    const double PI = 3.14159265358979323846;
    canvas.add(new Circle(2.0));
    double expectedTotal = 20.0 + (PI * 4.0);
    assert(areAlmostEqual(canvas.totalArea(), expectedTotal));

    // Base shape area: 0.0
    canvas.add(new Shape());
    assert(areAlmostEqual(canvas.totalArea(), expectedTotal));

    std::cout << "Test test_Canvas_totalArea: PASS" << std::endl;
}

// Test largestArea calculation
void test_Canvas_largestArea() {
    Canvas canvas;
    
    // Empty canvas
    assert(areAlmostEqual(canvas.largestArea(), 0.0));

    // Add rectangle with area 10.0 (2 * 5)
    canvas.add(new Rectangle(2.0, 5.0));
    assert(areAlmostEqual(canvas.largestArea(), 10.0));

    // Add rectangle with larger area 50.0 (5 * 10)
    canvas.add(new Rectangle(5.0, 10.0));
    assert(areAlmostEqual(canvas.largestArea(), 50.0));

    // Add circle with smaller area ~28.27 (PI * 3^2)
    canvas.add(new Circle(3.0));
    assert(areAlmostEqual(canvas.largestArea(), 50.0));

    // Add circle with larger area ~113.10 (PI * 6^2)
    const double PI = 3.14159265358979323846;
    canvas.add(new Circle(6.0));
    assert(areAlmostEqual(canvas.largestArea(), PI * 36.0));

    std::cout << "Test test_Canvas_largestArea: PASS" << std::endl;
}

// Test clear() functionality
void test_Canvas_clear() {
    Canvas canvas;
    canvas.add(new Rectangle(2.0, 2.0));
    canvas.add(new Circle(3.0));
    assert(canvas.count() == 2);

    canvas.clear();
    assert(canvas.count() == 0);
    assert(canvas.shapes.empty());
    assert(areAlmostEqual(canvas.totalArea(), 0.0));
    assert(areAlmostEqual(canvas.largestArea(), 0.0));

    // Ensure we can add again after clearing
    canvas.add(new Rectangle(1.0, 1.0));
    assert(canvas.count() == 1);
    assert(areAlmostEqual(canvas.totalArea(), 1.0));
    assert(areAlmostEqual(canvas.largestArea(), 1.0));

    std::cout << "Test test_Canvas_clear: PASS" << std::endl;
}

// Test multiple clears and empty clear
void test_Canvas_multipleClear() {
    Canvas canvas;
    // Clearing an empty canvas should be safe
    canvas.clear();
    assert(canvas.count() == 0);

    canvas.add(new Rectangle(2.0, 2.0));
    canvas.clear();
    assert(canvas.count() == 0);

    // Second clear on emptied canvas
    canvas.clear();
    assert(canvas.count() == 0);

    std::cout << "Test test_Canvas_multipleClear: PASS" << std::endl;
}

int main() {
    std::cout << "Running Canvas unit tests..." << std::endl;

    test_Canvas_initialState();
    test_Canvas_addNull();
    test_Canvas_addShapesAndCount();
    test_Canvas_totalArea();
    test_Canvas_largestArea();
    test_Canvas_clear();
    test_Canvas_multipleClear();

    std::cout << "All Canvas unit tests passed successfully!" << std::endl;
    return 0;
}
```
