Here is the complete C++ unit test suite covering `SamplePipeline` and related integration points.

```cpp
#include <iostream>
#include <cassert>
#include <cmath>
#include <vector>
#include <string>

#include "SamplePipeline.h"
#include "Filter.h"
#include "Sample.h"
#include "UnitScale.h"
#include "Channel.h"

// Helper function for floating point comparisons
bool doubleNear(double a, double b, double epsilon = 1e-6) {
    return std::fabs(a - b) <= epsilon;
}

void test_constructors_and_defaults() {
    // Default constructor
    SamplePipeline defaultPipeline;
    assert(defaultPipeline.acceptedCount() == 0);
    assert(defaultPipeline.rejectedCount() == 0);
    assert(defaultPipeline.filterCount() == 0);
    assert(defaultPipeline.minimumQuality == 1);
    assert(defaultPipeline.channel.channelName() == "unnamed");
    assert(defaultPipeline.channel.capacity == 16);
    assert(defaultPipeline.channel.isEmpty());
    assert(defaultPipeline.describe() == "empty");

    // Custom constructor
    SamplePipeline customPipeline("temp_sensor", 5);
    assert(customPipeline.acceptedCount() == 0);
    assert(customPipeline.rejectedCount() == 0);
    assert(customPipeline.filterCount() == 0);
    assert(customPipeline.minimumQuality == 1);
    assert(customPipeline.channel.channelName() == "temp_sensor");
    assert(customPipeline.channel.capacity == 5);
    assert(customPipeline.channel.isEmpty());
    assert(customPipeline.describe() == "empty");

    std::cout << "Test test_constructors_and_defaults: PASS" << std::endl;
}

void test_filter_management() {
    SamplePipeline pipeline("ch1", 10);
    assert(pipeline.filterCount() == 0);

    // Adding null filter should not alter filter count
    int count = pipeline.addFilter(nullptr);
    assert(count == 0);
    assert(pipeline.filterCount() == 0);

    PassFilter pf;
    ClampFilter cf(0.0, 10.0);
    SmoothFilter sf(0.5);

    assert(pipeline.addFilter(&pf) == 1);
    assert(pipeline.addFilter(&cf) == 2);
    assert(pipeline.addFilter(&sf) == 3);
    assert(pipeline.filterCount() == 3);

    // Clear filters
    int cleared = pipeline.clearFilters();
    assert(cleared == 3);
    assert(pipeline.filterCount() == 0);

    // Clear already empty filters
    assert(pipeline.clearFilters() == 0);

    std::cout << "Test test_filter_management: PASS" << std::endl;
}

void test_process_and_stage() {
    SamplePipeline pipeline("sensor", 10);
    ClampFilter cf(0.0, 100.0);
    SmoothFilter sf(0.5);

    pipeline.addFilter(&cf);
    pipeline.addFilter(&sf);

    // Test stage execution
    // Stage 0: ClampFilter on 150 -> 100
    double s0 = pipeline.stage(150.0, 0);
    assert(doubleNear(s0, 100.0));

    // Out of bound stages should return original reading
    assert(doubleNear(pipeline.stage(42.0, -1), 42.0));
    assert(doubleNear(pipeline.stage(42.0, 2), 42.0));

    // Reset filters for a clean sequence
    SamplePipeline pipe2("seq", 10);
    ClampFilter cf2(0.0, 50.0);      // Clamps > 50 to 50
    SmoothFilter sf2(0.5);          // First call primes value
    pipe2.addFilter(&cf2);
    pipe2.addFilter(&sf2);

    // 1st process: 100 -> clamped to 50 -> smooth primed with 50 -> 50
    double out1 = pipe2.process(100.0);
    assert(doubleNear(out1, 50.0));

    // 2nd process: 10 -> clamped to 10 -> smooth: 0.5*10 + 0.5*50 = 30
    double out2 = pipe2.process(10.0);
    assert(doubleNear(out2, 30.0));

    std::cout << "Test test_process_and_stage: PASS" << std::endl;
}

void test_wouldReject() {
    SamplePipeline pipeline("sensor", 10);
    ClampFilter cf(0.0, 50.0);
    pipeline.addFilter(&cf);

    // ClampFilter rejects if reading < low || reading > high
    assert(pipeline.wouldReject(25.0) == false);
    assert(pipeline.wouldReject(-1.0) == true);
    assert(pipeline.wouldReject(50.1) == true);

    Filter genericFilter;
    genericFilter.toggle(); // enabled = false
    SamplePipeline pipeDisabled("sensor", 10);
    pipeDisabled.addFilter(&genericFilter);
    // When disabled, rejects non-zero readings
    assert(pipeDisabled.wouldReject(10.0) == true);
    assert(pipeDisabled.wouldReject(0.0) == false);

    std::cout << "Test test_wouldReject: PASS" << std::endl;
}

void test_quality_and_rejection() {
    SamplePipeline pipeline("ch", 5);
    
    // Default min quality is 1
    assert(pipeline.requireQuality(50) == 1); // returns previous (1)
    assert(pipeline.minimumQuality == 50);

    // Negative requireQuality should not change setting
    assert(pipeline.requireQuality(-5) == 50);
    assert(pipeline.minimumQuality == 50);

    // Ingest sample with quality below minimum
    int res = pipeline.ingest(1, 20.0, 49);
    assert(res == -1);
    assert(pipeline.acceptedCount() == 0);
    assert(pipeline.rejectedCount() == 1);

    // Ingest sample meeting quality requirement
    res = pipeline.ingest(2, 20.0, 50);
    assert(res == 1);
    assert(pipeline.acceptedCount() == 1);
    assert(pipeline.rejectedCount() == 1);

    // Add filter that rejects values > 100
    ClampFilter cf(0.0, 100.0);
    pipeline.addFilter(&cf);

    res = pipeline.ingest(3, 150.0, 80); // rejected by filter
    assert(res == -1);
    assert(pipeline.acceptedCount() == 1);
    assert(pipeline.rejectedCount() == 2);

    std::cout << "Test test_quality_and_rejection: PASS" << std::endl;
}

void test_capacity_and_saturation() {
    SamplePipeline pipeline("small_channel", 2);
    assert(!pipeline.isSaturated());

    assert(pipeline.ingest(1, 10.0, 100) == 1);
    assert(!pipeline.isSaturated());
    assert(pipeline.validSamples() == 1);

    assert(pipeline.ingest(2, 20.0, 100) == 2);
    assert(pipeline.isSaturated());
    assert(pipeline.validSamples() == 2);

    // Next ingest exceeds capacity
    int res = pipeline.ingest(3, 30.0, 100);
    assert(res == -1);
    assert(pipeline.acceptedCount() == 2);
    assert(pipeline.rejectedCount() == 1);
    assert(pipeline.channel.droppedCount() == 1);

    std::cout << "Test test_capacity_and_saturation: PASS" << std::endl;
}

void test_ingestBatch() {
    SamplePipeline pipeline("batch_channel", 4);
    std::vector<double> readings = { 10.0, 20.0, 30.0, 40.0, 50.0 };

    int taken = pipeline.ingestBatch(readings, 80);
    // Capacity is 4, so first 4 accepted, 5th rejected
    assert(taken == 4);
    assert(pipeline.acceptedCount() == 4);
    assert(pipeline.rejectedCount() == 1);
    assert(pipeline.isSaturated());
    assert(pipeline.validSamples() == 4);

    std::cout << "Test test_ingestBatch: PASS" << std::endl;
}

void test_channel_metrics_and_latest() {
    SamplePipeline pipeline("metrics_chan", 5);
    pipeline.ingest(10, 15.0, 90);
    pipeline.ingest(11, 25.0, 90);
    pipeline.ingest(12, 5.0, 90);

    assert(doubleNear(pipeline.channelMean(), 15.0));
    assert(doubleNear(pipeline.channelPeak(), 25.0));

    Sample lat = pipeline.latest();
    assert(lat.position() == 12);
    assert(doubleNear(lat.value(), 5.0));
    assert(lat.score() == 90);
    assert(lat.isValid());

    std::cout << "Test test_channel_metrics_and_latest: PASS" << std::endl;
}

void test_unit_scale_normalization_and_describe() {
    SamplePipeline pipeline("temp_c", 5);
    // UnitScale(factor, offset, unitName)
    // toBase = reading * 1.8 + 32.0 (C to F)
    // fromBase = (reading - 32.0) / 1.8 (F to C)
    pipeline.scale = UnitScale(1.8, 32.0, "degC");

    assert(pipeline.describe() == "empty");

    pipeline.ingest(1, 100.0, 80);
    assert(pipeline.describe() == "temp_c:degC");

    // Normalised uses scale.fromBase()
    // (212 - 32) / 1.8 = 100
    assert(doubleNear(pipeline.normalised(212.0), 100.0));

    // Denormalised uses scale.toBase()
    // 100 * 1.8 + 32 = 212
    assert(doubleNear(pipeline.denormalised(100.0), 212.0));

    std::cout << "Test test_unit_scale_normalization_and_describe: PASS" << std::endl;
}

void test_flush() {
    SamplePipeline pipeline("flush_chan", 3);
    pipeline.ingest(1, 10.0, 90);
    pipeline.ingest(2, 20.0, 90);
    pipeline.ingest(3, 30.0, 0); // Invalid quality, rejected

    assert(pipeline.acceptedCount() == 2);
    assert(pipeline.rejectedCount() == 1);
    assert(pipeline.validSamples() == 2);

    int previousCount = pipeline.flush();
    assert(previousCount == 2);
    assert(pipeline.acceptedCount() == 0);
    assert(pipeline.rejectedCount() == 0);
    assert(pipeline.validSamples() == 0);
    assert(pipeline.channel.isEmpty());
    assert(pipeline.channel.capacity == 3);
    assert(pipeline.channel.channelName() == "flush_chan");
    assert(pipeline.describe() == "empty");

    std::cout << "Test test_flush: PASS" << std::endl;
}

int main() {
    std::cout << "--- Starting SamplePipeline Test Suite ---" << std::endl;

    test_constructors_and_defaults();
    test_filter_management();
    test_process_and_stage();
    test_wouldReject();
    test_quality_and_rejection();
    test_capacity_and_saturation();
    test_ingestBatch();
    test_channel_metrics_and_latest();
    test_unit_scale_normalization_and_describe();
    test_flush();

    std::cout << "--- All SamplePipeline Tests Passed! ---" << std::endl;
    return 0;
}
```
