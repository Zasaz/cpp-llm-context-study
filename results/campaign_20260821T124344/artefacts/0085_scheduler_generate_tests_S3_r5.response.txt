Here is the complete C++ unit test suite covering `TaskQueue.cpp` and its interactions with `Policy` and `Task`.

```cpp
#include <iostream>
#include <cassert>
#include <string>
#include <vector>

#include "TaskQueue.h"
#include "Policy.h"
#include "Task.h"

// Test default and parameterized constructors, and initial states
void test_constructors_and_defaults() {
    TaskQueue defaultQueue;
    assert(defaultQueue.capacity == 8);
    assert(defaultQueue.policy == nullptr);
    assert(defaultQueue.rejectedCount() == 0);
    assert(defaultQueue.size() == 0);
    assert(defaultQueue.isEmpty());
    assert(!defaultQueue.isFull());
    assert(defaultQueue.bestIndex() == -1);

    Task bestEmpty = defaultQueue.best();
    assert(bestEmpty.identifier() == 0);
    assert(bestEmpty.label() == "unnamed");

    Task dequeuedEmpty = defaultQueue.dequeue();
    assert(dequeuedEmpty.identifier() == 0);

    // Parameterized constructor with positive limit
    TaskQueue customQueue(3);
    assert(customQueue.capacity == 3);
    assert(customQueue.rejectedCount() == 0);

    // Parameterized constructor with limit <= 0 (should clamp to 1)
    TaskQueue zeroQueue(0);
    assert(zeroQueue.capacity == 1);

    TaskQueue negativeQueue(-5);
    assert(negativeQueue.capacity == 1);

    std::cout << "Test test_constructors_and_defaults: PASS" << std::endl;
}

// Test basic enqueue and capacity limits
void test_enqueue_and_capacity() {
    TaskQueue queue(2);
    Task t1(1, "Task1", 5, 10);
    Task t2(2, "Task2", 3, 20);
    Task t3(3, "Task3", 8, 15);

    int res1 = queue.enqueue(t1);
    assert(res1 == 1);
    assert(queue.size() == 1);
    assert(!queue.isEmpty());
    assert(!queue.isFull());

    int res2 = queue.enqueue(t2);
    assert(res2 == 2);
    assert(queue.size() == 2);
    assert(queue.isFull());

    // Should reject because queue is full
    int res3 = queue.enqueue(t3);
    assert(res3 == -1);
    assert(queue.size() == 2);
    assert(queue.rejectedCount() == 1);

    // Try enqueueing another rejected task
    queue.enqueue(t3);
    assert(queue.rejectedCount() == 2);

    std::cout << "Test test_enqueue_and_capacity: PASS" << std::endl;
}

// Test policy assignment and rejection based on policy rules
void test_policy_rejections() {
    TaskQueue queue(5);
    PriorityPolicy priorityPolicy(10); // rejects priority < 10

    queue.usePolicy(&priorityPolicy);
    assert(queue.policy == &priorityPolicy);

    Task lowPriorityTask(1, "Low", 5, 10);
    Task highPriorityTask(2, "High", 12, 10);

    int res1 = queue.enqueue(lowPriorityTask);
    assert(res1 == -1);
    assert(queue.rejectedCount() == 1);
    assert(queue.size() == 0);

    int res2 = queue.enqueue(highPriorityTask);
    assert(res2 == 1);
    assert(queue.rejectedCount() == 1);
    assert(queue.size() == 1);

    // Test with DeadlinePolicy
    DeadlinePolicy deadlinePolicy(50); // rejects deadline > 50
    queue.usePolicy(&deadlinePolicy);

    Task longDeadlineTask(3, "Long", 5, 10);
    longDeadlineTask.setDeadline(100);
    int res3 = queue.enqueue(longDeadlineTask);
    assert(res3 == -1);
    assert(queue.rejectedCount() == 2);

    Task shortDeadlineTask(4, "Short", 5, 10);
    shortDeadlineTask.setDeadline(30);
    int res4 = queue.enqueue(shortDeadlineTask);
    assert(res4 == 2);
    assert(queue.rejectedCount() == 2);

    std::cout << "Test test_policy_rejections: PASS" << std::endl;
}

// Test selection and dequeueing without a policy (uses Task::weight())
void test_default_ranking_and_dequeue() {
    TaskQueue queue(5);
    Task t1(1, "Task1", 2, 10); // weight: 2 * 10 = 20
    Task t2(2, "Task2", 8, 10); // weight: 8 * 10 = 80
    Task t3(3, "Task3", 4, 10); // weight: 4 * 10 = 40

    queue.enqueue(t1);
    queue.enqueue(t2);
    queue.enqueue(t3);

    assert(queue.bestIndex() == 1);
    assert(queue.best().identifier() == 2);

    Task dequeued = queue.dequeue();
    assert(dequeued.identifier() == 2);
    assert(queue.size() == 2);

    // Next best should now be t3
    assert(queue.bestIndex() == 1);
    assert(queue.best().identifier() == 3);

    // Drain the rest
    Task dequeued2 = queue.dequeue();
    assert(dequeued2.identifier() == 3);
    Task dequeued3 = queue.dequeue();
    assert(dequeued3.identifier() == 1);

    assert(queue.isEmpty());
    assert(queue.bestIndex() == -1);

    std::cout << "Test test_default_ranking_and_dequeue: PASS" << std::endl;
}

// Test ranking and selection with various policies (FifoPolicy, PriorityPolicy, DeadlinePolicy)
void test_policy_ranking() {
    TaskQueue queue(5);
    FifoPolicy fifoPolicy(2); // rank = bias - id * 2 (smaller ID => higher rank)
    queue.usePolicy(&fifoPolicy);

    Task t10(10, "Task10", 1, 10);
    Task t5(5, "Task5", 1, 10);
    Task t8(8, "Task8", 1, 10);

    queue.enqueue(t10);
    queue.enqueue(t5);
    queue.enqueue(t8);

    // For t10: -20, t5: -10, t8: -16 -> highest score is t5 (index 1)
    assert(queue.bestIndex() == 1);
    assert(queue.best().identifier() == 5);

    // Switch to PriorityPolicy
    PriorityPolicy priorityPolicy(0); // rank = priority * 100 - duration
    queue.usePolicy(&priorityPolicy);

    // t10 score: 100 - 10 = 90
    // t5 score: 100 - 10 = 90
    // First max encountered remains index 0
    assert(queue.bestIndex() == 0);

    Task tHigh(1, "HighPriority", 5, 20); // score: 500 - 20 = 480
    queue.enqueue(tHigh);
    assert(queue.bestIndex() == 3);
    assert(queue.best().identifier() == 1);

    std::cout << "Test test_policy_ranking: PASS" << std::endl;
}

// Test remaining work calculations and pending task count
void test_metrics_and_accounting() {
    TaskQueue queue(5);
    Task t1(1, "Task1", 1, 10);
    Task t2(2, "Task2", 1, 15);
    Task t3(3, "Task3", 1, 5);

    queue.enqueue(t1);
    queue.enqueue(t2);
    queue.enqueue(t3);

    assert(queue.totalRemaining() == 30);
    assert(queue.pendingCount() == 3);

    // Start one task
    queue.tasks[0].advance(4); // remaining: 6, started: true
    assert(queue.totalRemaining() == 26);
    assert(queue.pendingCount() == 2);

    std::cout << "Test test_metrics_and_accounting: PASS" << std::endl;
}

// Test advanceAll and purgeComplete functionality
void test_advance_all_and_purge() {
    TaskQueue queue(5);
    Task t1(1, "Task1", 1, 10);
    Task t2(2, "Task2", 1, 5);
    Task t3(3, "Task3", 1, 15);

    queue.enqueue(t1);
    queue.enqueue(t2);
    queue.enqueue(t3);

    // Advance all by 6 units
    // t1: elapsed = 6, remaining = 4, complete = false
    // t2: elapsed = 5 (capped), remaining = 0, complete = true
    // t3: elapsed = 6, remaining = 9, complete = false
    int completedCount = queue.advanceAll(6);
    assert(completedCount == 1);
    assert(queue.totalRemaining() == 13);
    assert(queue.pendingCount() == 0);

    // Purge completed tasks
    int purgedCount = queue.purgeComplete();
    assert(purgedCount == 1);
    assert(queue.size() == 2);
    assert(queue.tasks[0].identifier() == 1);
    assert(queue.tasks[1].identifier() == 3);

    // Advance remaining tasks until completion
    completedCount = queue.advanceAll(20);
    assert(completedCount == 2);
    assert(queue.totalRemaining() == 0);

    purgedCount = queue.purgeComplete();
    assert(purgedCount == 2);
    assert(queue.isEmpty());
    assert(queue.size() == 0);

    // Purge on empty queue
    purgedCount = queue.purgeComplete();
    assert(purgedCount == 0);

    std::cout << "Test test_advance_all_and_purge: PASS" << std::endl;
}

int main() {
    test_constructors_and_defaults();
    test_enqueue_and_capacity();
    test_policy_rejections();
    test_default_ranking_and_dequeue();
    test_policy_ranking();
    test_metrics_and_accounting();
    test_advance_all_and_purge();

    std::cout << "\nAll TaskQueue tests passed successfully!" << std::endl;
    return 0;
}
```
