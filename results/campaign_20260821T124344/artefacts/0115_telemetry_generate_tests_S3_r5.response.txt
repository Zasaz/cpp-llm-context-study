Here is the complete C++ unit test suite covering the `SamplePipeline` class and its interactions with the associated classes.

```cpp
#include <iostream>
#include <cassert>
#include <cmath>
#include <vector>
#include <string>

#include "SamplePipeline.h"
#include "Channel.h"
#include "Filter.h"
#include "Sample.h"
#include "UnitScale.h"

// Helper function for floating point comparison
bool are_close(double a, double b, double epsilon = 1e-6) {
    return std::fabs(a - b) < epsilon;
}

// 1. Test Default and Parameterized Constructors
void test_constructors() {
    SamplePipeline pipelineDefault;
    assert(pipelineDefault.acceptedCount() == 0);
    assert(pipelineDefault.rejectedCount() == 0);
    assert(pipelineDefault.minimumQuality == 1);
    assert(pipelineDefault.filterCount() == 0);
    assert(pipelineDefault.channel.channelName() == "unnamed");
    assert(pipelineDefault.channel.capacity == 16);

    SamplePipeline pipelineCustom("sensor_voltage", 5);
    assert(pipelineCustom.acceptedCount() == 0);
    assert(pipelineCustom.rejectedCount() == 0);
    assert(pipelineCustom.minimumQuality == 1);
    assert(pipelineCustom.filterCount() == 0);
    assert(pipelineCustom.channel.channelName() == "sensor_voltage");
    assert(pipelineCustom.channel.capacity == 5);

    std::cout << "Test test_constructors: PASS" << std::endl;
}

// 2. Test Filter Management (Add, Count, Clear, Stage)
void test_filter_management() {
    SamplePipeline pipeline;
    ClampFilter clamp(0.0, 10.0);
    SmoothFilter smooth(0.5);

    // Null filter check
    assert(pipeline.addFilter(nullptr) == 0);
    assert(pipeline.filterCount() == 0);

    // Adding valid filters
    assert(pipeline.addFilter(&clamp) == 1);
    assert(pipeline.filterCount() == 1);
    assert(pipeline.addFilter(&smooth) == 2);
    assert(pipeline.filterCount() == 2);

    // Stage processing bounds and execution
    assert(are_close(pipeline.stage(15.0, 0), 10.0)); // stage 0 is clamp
    assert(are_close(pipeline.stage(15.0, -1), 15.0)); // out of bounds low
    assert(are_close(pipeline.stage(15.0, 2), 15.0));  // out of bounds high

    // Clear filters
    int previousCount = pipeline.clearFilters();
    assert(previousCount == 2);
    assert(pipeline.filterCount() == 0);

    std::cout << "Test test_filter_management: PASS" << std::endl;
}

// 3. Test Process Chain
void test_process_chain() {
    SamplePipeline pipeline;
    ClampFilter clamp(0.0, 100.0);
    SmoothFilter smooth(0.5);

    pipeline.addFilter(&clamp);
    pipeline.addFilter(&smooth);

    // First value: 150 -> clamped to 100 -> smooth primed with 100
    double out1 = pipeline.process(150.0);
    assert(are_close(out1, 100.0));

    // Second value: -50 -> clamped to 0 -> smooth: 0.5 * 0 + 0.5 * 100 = 50.0
    double out2 = pipeline.process(-50.0);
    assert(are_close(out2, 50.0));

    std::cout << "Test test_process_chain: PASS" << std::endl;
}

// 4. Test Ingestion and Filtering Logic
void test_ingest_basic_and_filters() {
    SamplePipeline pipeline("temp_sensor", 10);
    ClampFilter clamp(0.0, 50.0);
    pipeline.addFilter(&clamp);

    // Ingest within limits
    int res1 = pipeline.ingest(1, 25.0, 80);
    assert(res1 == 1);
    assert(pipeline.acceptedCount() == 1);
    assert(pipeline.rejectedCount() == 0);
    assert(pipeline.validSamples() == 1);
    assert(are_close(pipeline.latest().value(), 25.0));
    assert(pipeline.latest().position() == 1);

    // Ingest value that gets rejected by wouldReject (clamp rejects > 50.0)
    int res2 = pipeline.ingest(2, 60.0, 80);
    assert(res2 == -1);
    assert(pipeline.acceptedCount() == 1);
    assert(pipeline.rejectedCount() == 1);

    // Ingest value that gets rejected by quality
    int res3 = pipeline.ingest(3, 30.0, 0); // Default min quality is 1
    assert(res3 == -1);
    assert(pipeline.acceptedCount() == 1);
    assert(pipeline.rejectedCount() == 2);

    std::cout << "Test test_ingest_basic_and_filters: PASS" << std::endl;
}

// 5. Test Quality Requirements
void test_require_quality() {
    SamplePipeline pipeline;
    int prev = pipeline.requireQuality(75);
    assert(prev == 1);
    assert(pipeline.minimumQuality == 75);

    // Negative quality should be ignored
    pipeline.requireQuality(-5);
    assert(pipeline.minimumQuality == 75);

    // Ingest below quality
    int res1 = pipeline.ingest(1, 10.0, 74);
    assert(res1 == -1);
    assert(pipeline.rejectedCount() == 1);

    // Ingest at or above quality
    int res2 = pipeline.ingest(2, 10.0, 75);
    assert(res2 == 1);
    assert(pipeline.acceptedCount() == 1);

    std::cout << "Test test_require_quality: PASS" << std::endl;
}

// 6. Test Capacity and Saturation
void test_capacity_and_saturation() {
    SamplePipeline pipeline("ch1", 2);
    assert(!pipeline.isSaturated());

    assert(pipeline.ingest(1, 10.0, 80) == 1);
    assert(!pipeline.isSaturated());

    assert(pipeline.ingest(2, 20.0, 80) == 2);
    assert(pipeline.isSaturated());

    // Ingest on full pipeline should fail
    int res = pipeline.ingest(3, 30.0, 80);
    assert(res == -1);
    assert(pipeline.acceptedCount() == 2);
    assert(pipeline.rejectedCount() == 1);
    assert(pipeline.isSaturated());

    std::cout << "Test test_capacity_and_saturation: PASS" << std::endl;
}

// 7. Test Batch Ingestion
void test_ingest_batch() {
    SamplePipeline pipeline("ch_batch", 10);
    ClampFilter clamp(0.0, 100.0);
    pipeline.addFilter(&clamp);

    std::vector<double> batch = {10.0, 20.0, 150.0, 40.0}; // 150.0 will be rejected by clamp
    int taken = pipeline.ingestBatch(batch, 90);

    assert(taken == 3);
    assert(pipeline.acceptedCount() == 3);
    assert(pipeline.rejectedCount() == 1);
    assert(pipeline.validSamples() == 3);

    std::cout << "Test test_ingest_batch: PASS" << std::endl;
}

// 8. Test Statistics and Scale Transformations
void test_statistics_and_scaling() {
    SamplePipeline pipeline("sensor", 10);
    pipeline.scale = UnitScale(2.0, 5.0, "mV"); // toBase: x * 2 + 5; fromBase: (x - 5) / 2

    pipeline.ingest(1, 10.0, 90);
    pipeline.ingest(2, 20.0, 90);
    pipeline.ingest(3, 30.0, 90);

    // Channel stats
    assert(are_close(pipeline.channelMean(), 20.0));
    assert(are_close(pipeline.channelPeak(), 30.0));

    // Normalisation / Denormalisation
    // Denormalised (toBase): 10 * 2 + 5 = 25
    assert(are_close(pipeline.denormalised(10.0), 25.0));
    // Normalised (fromBase): (25 - 5) / 2 = 10
    assert(are_close(pipeline.normalised(25.0), 10.0));

    std::cout << "Test test_statistics_and_scaling: PASS" << std::endl;
}

// 9. Test Flush and Describe
void test_flush_and_describe() {
    SamplePipeline pipeline("pressure", 5);
    pipeline.scale = UnitScale(1.0, 0.0, "psi");

    // Initially empty
    assert(pipeline.describe() == "empty");

    pipeline.ingest(1, 14.7, 100);
    assert(pipeline.describe() == "pressure:psi");

    int clearedSamples = pipeline.flush();
    assert(clearedSamples == 1);
    assert(pipeline.acceptedCount() == 0);
    assert(pipeline.rejectedCount() == 0);
    assert(pipeline.channel.isEmpty());
    assert(pipeline.describe() == "empty");

    std::cout << "Test test_flush_and_describe: PASS" << std::endl;
}

int main() {
    std::cout << "Starting SamplePipeline Test Suite..." << std::endl;

    test_constructors();
    test_filter_management();
    test_process_chain();
    test_ingest_basic_and_filters();
    test_require_quality();
    test_capacity_and_saturation();
    test_ingest_batch();
    test_statistics_and_scaling();
    test_flush_and_describe();

    std::cout << "All SamplePipeline tests passed successfully!" << std::endl;
    return 0;
}
```
