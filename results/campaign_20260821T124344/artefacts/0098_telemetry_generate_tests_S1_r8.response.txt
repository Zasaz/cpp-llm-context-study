#include <iostream>
#include <cassert>
#include <cmath>
#include <vector>
#include <string>

#include "SamplePipeline.h"
#include "Filter.h"
#include "UnitScale.h"
#include "Sample.h"
#include "Channel.h"

// Helper function for floating-point comparison
bool approxEqual(double a, double b, double epsilon = 1e-6) {
    return std::fabs(a - b) < epsilon;
}

// Test constructors and default state of SamplePipeline
void test_SamplePipeline_constructors() {
    SamplePipeline defaultPipe;
    assert(defaultPipe.acceptedCount() == 0);
    assert(defaultPipe.rejectedCount() == 0);
    assert(defaultPipe.filterCount() == 0);
    assert(defaultPipe.minimumQuality == 1);
    assert(defaultPipe.channel.channelName() == "unnamed");
    assert(defaultPipe.channel.capacity == 16);
    assert(defaultPipe.isSaturated() == false);
    assert(defaultPipe.validSamples() == 0);
    assert(defaultPipe.describe() == "empty");

    SamplePipeline paramPipe("temperature", 5);
    assert(paramPipe.acceptedCount() == 0);
    assert(paramPipe.rejectedCount() == 0);
    assert(paramPipe.filterCount() == 0);
    assert(paramPipe.minimumQuality == 1);
    assert(paramPipe.channel.channelName() == "temperature");
    assert(paramPipe.channel.capacity == 5);
    assert(paramPipe.describe() == "empty");

    std::cout << "Test test_SamplePipeline_constructors: PASS" << std::endl;
}

// Test filter management: addFilter, filterCount, clearFilters
void test_SamplePipeline_filterManagement() {
    SamplePipeline pipe("test", 10);
    assert(pipe.filterCount() == 0);

    // Null pointer filter check
    int countAfterNull = pipe.addFilter(nullptr);
    assert(countAfterNull == 0);
    assert(pipe.filterCount() == 0);

    PassFilter pass;
    ClampFilter clamp(0.0, 100.0);
    SmoothFilter smooth(0.5);

    assert(pipe.addFilter(&pass) == 1);
    assert(pipe.addFilter(&clamp) == 2);
    assert(pipe.addFilter(&smooth) == 3);
    assert(pipe.filterCount() == 3);

    int clearedCount = pipe.clearFilters();
    assert(clearedCount == 3);
    assert(pipe.filterCount() == 0);

    std::cout << "Test test_SamplePipeline_filterManagement: PASS" << std::endl;
}

// Test process and stage functions
void test_SamplePipeline_processAndStage() {
    SamplePipeline pipe("dsp", 10);
    ClampFilter clamp(10.0, 50.0);
    SmoothFilter smooth(0.5);

    pipe.addFilter(&clamp);
    pipe.addFilter(&smooth);

    // Out of range stage queries should return raw value
    assert(approxEqual(pipe.stage(5.0, -1), 5.0));
    assert(approxEqual(pipe.stage(5.0, 2), 5.0));

    // Test stage 0 (clamp) directly
    assert(approxEqual(pipe.stage(5.0, 0), 10.0));
    assert(approxEqual(pipe.stage(60.0, 0), 50.0));
    assert(approxEqual(pipe.stage(25.0, 0), 25.0));

    // Pipeline process: 1st reading (5.0) clamped to 10.0 -> smooth unprimed sets 10.0
    double out1 = pipe.process(5.0);
    assert(approxEqual(out1, 10.0));

    // 2nd reading (100.0) clamped to 50.0 -> smooth: 0.5 * 50.0 + 0.5 * 10.0 = 30.0
    double out2 = pipe.process(100.0);
    assert(approxEqual(out2, 30.0));

    std::cout << "Test test_SamplePipeline_processAndStage: PASS" << std::endl;
}

// Test wouldReject function
void test_SamplePipeline_wouldReject() {
    SamplePipeline pipe("reject_test", 10);
    ClampFilter clamp(0.0, 10.0);
    pipe.addFilter(&clamp);

    assert(!pipe.wouldReject(5.0));
    assert(!pipe.wouldReject(0.0));
    assert(!pipe.wouldReject(10.0));
    assert(pipe.wouldReject(-0.1));
    assert(pipe.wouldReject(10.1));

    // Disabled filter rejection behavior
    PassFilter pass;
    pass.toggle(); // enabled becomes false
    SamplePipeline pipe2("disabled_filter", 10);
    pipe2.addFilter(&pass);
    assert(pipe2.wouldReject(5.0));
    assert(!pipe2.wouldReject(0.0)); // Filter::rejects is false if reading == 0.0

    std::cout << "Test test_SamplePipeline_wouldReject: PASS" << std::endl;
}

// Test ingest logic, quality constraints, and rejection paths
void test_SamplePipeline_ingest() {
    SamplePipeline pipe("stream", 3);
    ClampFilter clamp(0.0, 100.0);
    pipe.addFilter(&clamp);

    // Initial requireQuality check
    int prevMin = pipe.requireQuality(10);
    assert(prevMin == 1);
    assert(pipe.minimumQuality == 10);
    // Ignore negative values in requireQuality
    pipe.requireQuality(-5);
    assert(pipe.minimumQuality == 10);

    // 1. Rejected due to quality < minimumQuality
    int res = pipe.ingest(1, 50.0, 5);
    assert(res == -1);
    assert(pipe.acceptedCount() == 0);
    assert(pipe.rejectedCount() == 1);

    // 2. Rejected due to wouldReject (clamped range is 0.0 to 100.0)
    res = pipe.ingest(2, 150.0, 50);
    assert(res == -1);
    assert(pipe.rejectedCount() == 2);

    // 3. Rejected due to invalid sample creation (quality <= 0)
    pipe.requireQuality(0);
    res = pipe.ingest(3, 50.0, 0); // Sample with quality 0 is flagged/not valid
    assert(res == -1);
    assert(pipe.rejectedCount() == 3);

    // 4. Successful ingests up to capacity (capacity = 3)
    res = pipe.ingest(101, 10.0, 80);
    assert(res == 1);
    assert(pipe.acceptedCount() == 1);

    res = pipe.ingest(102, 20.0, 90);
    assert(res == 2);
    assert(pipe.acceptedCount() == 2);

    res = pipe.ingest(103, 30.0, 85);
    assert(res == 3);
    assert(pipe.acceptedCount() == 3);
    assert(pipe.isSaturated() == true);
    assert(pipe.validSamples() == 3);

    // 5. Rejected due to saturated channel
    res = pipe.ingest(104, 40.0, 95);
    assert(res == -1);
    assert(pipe.rejectedCount() == 4);
    assert(pipe.acceptedCount() == 3);

    std::cout << "Test test_SamplePipeline_ingest: PASS" << std::endl;
}

// Test ingestBatch
void test_SamplePipeline_ingestBatch() {
    SamplePipeline pipe("batch", 5);
    ClampFilter clamp(0.0, 50.0);
    pipe.addFilter(&clamp);

    std::vector<double> batch = { 10.0, 20.0, 60.0 /* rejected */, 30.0, 40.0, 50.0 /* rejected: full */ };
    int ingestedCount = pipe.ingestBatch(batch, 80);

    assert(ingestedCount == 4); // 10.0, 20.0, 30.0, 40.0 accepted
    assert(pipe.acceptedCount() == 4);
    assert(pipe.rejectedCount() == 2); // 60.0 (rejected by filter), 50.0 (rejected by capacity full: capacity is 5, but reached capacity on 4 items? Wait, 4 items in channel, 5th attempt: batch[4]=40.0 -> 4 items. batch[5]=50.0 -> 5 items fits!)

    // Let's verify capacity: limit was 5.
    // 10.0 -> accepted (1)
    // 20.0 -> accepted (2)
    // 60.0 -> rejected (out of clamp)
    // 30.0 -> accepted (3)
    // 40.0 -> accepted (4)
    // 50.0 -> accepted (5)
    // All 4 valid within bounds fit in capacity 5. Total accepted = 5.
    // Let's test with fresh pipeline and exact bounds.
    SamplePipeline batchPipe("batch2", 3);
    std::vector<double> items = { 5.0, 10.0, 15.0, 20.0 };
    int accepted = batchPipe.ingestBatch(items, 100);
    assert(accepted == 3);
    assert(batchPipe.acceptedCount() == 3);
    assert(batchPipe.rejectedCount() == 1);

    std::cout << "Test test_SamplePipeline_ingestBatch: PASS" << std::endl;
}

// Test metrics, statistics and latest sample
void test_SamplePipeline_statisticsAndLatest() {
    SamplePipeline pipe("sensor", 10);
    pipe.ingest(1, 10.0, 90);
    pipe.ingest(2, 20.0, 90);
    pipe.ingest(3, 30.0, 90);

    assert(approxEqual(pipe.channelMean(), 20.0));
    assert(approxEqual(pipe.channelPeak(), 30.0));

    Sample latestSample = pipe.latest();
    assert(latestSample.position() == 3);
    assert(approxEqual(latestSample.value(), 30.0));
    assert(latestSample.score() == 90);

    std::cout << "Test test_SamplePipeline_statisticsAndLatest: PASS" << std::endl;
}

// Test normalisation and denormalisation with UnitScale
void test_SamplePipeline_scaling() {
    SamplePipeline pipe("pressure", 10);
    pipe.scale = UnitScale(2.0, 10.0, "bar"); // base = reading * 2.0 + 10.0

    // normalised: reading in base -> user unit = (reading - 10.0) / 2.0
    double norm = pipe.normalised(30.0);
    assert(approxEqual(norm, 10.0));

    // denormalised: reading in user unit -> base = reading * 2.0 + 10.0
    double denorm = pipe.denormalised(10.0);
    assert(approxEqual(denorm, 30.0));

    std::cout << "Test test_SamplePipeline_scaling: PASS" << std::endl;
}

// Test flush function and describe string
void test_SamplePipeline_flushAndDescribe() {
    SamplePipeline pipe("humidity", 5);
    pipe.scale = UnitScale(1.0, 0.0, "%");

    assert(pipe.describe() == "empty");

    pipe.ingest(1, 45.0, 80);
    assert(pipe.describe() == "humidity:%");

    assert(pipe.acceptedCount() == 1);
    assert(pipe.channel.count() == 1);

    int flushedSamples = pipe.flush();
    assert(flushedSamples == 1);
    assert(pipe.acceptedCount() == 0);
    assert(pipe.rejectedCount() == 0);
    assert(pipe.channel.count() == 0);
    assert(pipe.channel.capacity == 5);
    assert(pipe.channel.channelName() == "humidity");
    assert(pipe.describe() == "empty");

    std::cout << "Test test_SamplePipeline_flushAndDescribe: PASS" << std::endl;
}

    test_SamplePipeline_filterManagement();
    test_SamplePipeline_processAndStage();
    test_SamplePipeline_wouldReject();
    test_SamplePipeline_ingest();
    test_SamplePipeline_ingestBatch();
    test_SamplePipeline_statisticsAndLatest();
    test_SamplePipeline_scaling();
    test_SamplePipeline_flushAndDescribe();

    std::cout << "All SamplePipeline tests passed successfully!" << std::endl;
    return 0;
}
