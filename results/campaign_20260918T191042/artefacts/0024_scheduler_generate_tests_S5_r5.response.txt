Here is the complete C++ unit test suite covering `TaskQueue` and its interactions with `Policy` and `Task`.

```cpp
#include <cassert>
#include <iostream>
#include <vector>
#include <string>

#include "TaskQueue.h"
#include "Policy.h"
#include "Task.h"

// Test default and parameterized constructors
void test_constructors_and_initial_state() {
    // Default constructor
    TaskQueue q1;
    assert(q1.size() == 0);
    assert(q1.isEmpty() == true);
    assert(q1.isFull() == false);
    assert(q1.capacity == 8);
    assert(q1.rejectedCount() == 0);
    assert(q1.policy == nullptr);

    // Parameterized constructor with positive limit
    TaskQueue q2(3);
    assert(q2.size() == 0);
    assert(q2.isEmpty() == true);
    assert(q2.isFull() == false);
    assert(q2.capacity == 3);
    assert(q2.rejectedCount() == 0);

    // Parameterized constructor with non-positive limit (should clamp to 1)
    TaskQueue q3(0);
    assert(q3.capacity == 1);

    TaskQueue q4(-5);
    assert(q4.capacity == 1);

    std::cout << "Test test_constructors_and_initial_state: PASS" << std::endl;
}

// Test enqueuing tasks, full capacity behavior, and rejection counting
void test_enqueue_capacity_and_rejection() {
    TaskQueue queue(2);
    Task t1(1, "Task1", 5, 10);
    Task t2(2, "Task2", 3, 20);
    Task t3(3, "Task3", 8, 15);

    int res1 = queue.enqueue(t1);
    assert(res1 == 1);
    assert(queue.size() == 1);
    assert(!queue.isEmpty());
    assert(!queue.isFull());

    int res2 = queue.enqueue(t2);
    assert(res2 == 2);
    assert(queue.size() == 2);
    assert(!queue.isEmpty());
    assert(queue.isFull());

    // Queue is now full, enqueue should fail
    int res3 = queue.enqueue(t3);
    assert(res3 == -1);
    assert(queue.size() == 2);
    assert(queue.rejectedCount() == 1);

    int res4 = queue.enqueue(t3);
    assert(res4 == -1);
    assert(queue.size() == 2);
    assert(queue.rejectedCount() == 2);

    std::cout << "Test test_enqueue_capacity_and_rejection: PASS" << std::endl;
}

// Test policy assignment and policy acceptance rules during enqueue
void test_policy_filtering_on_enqueue() {
    TaskQueue queue(5);
    PriorityPolicy minPriorityPolicy(5); // Rejects tasks with priority < 5

    queue.usePolicy(&minPriorityPolicy);

    Task validTask(1, "Valid", 7, 10);
    Task invalidTask(2, "Invalid", 3, 10);

    int res1 = queue.enqueue(validTask);
    assert(res1 == 1);
    assert(queue.size() == 1);
    assert(queue.rejectedCount() == 0);

    int res2 = queue.enqueue(invalidTask);
    assert(res2 == -1);
    assert(queue.size() == 1);
    assert(queue.rejectedCount() == 1);

    // Switch policy to DeadlinePolicy
    DeadlinePolicy deadlinePolicy(50); // Accepts tasks with deadline <= 50
    queue.usePolicy(&deadlinePolicy);

    Task tightDeadlineTask(3, "Tight", 2, 20); // default deadline = duration = 20 <= 50
    Task farDeadlineTask(4, "Far", 2, 100);    // default deadline = duration = 100 > 50

    assert(queue.enqueue(tightDeadlineTask) == 2);
    assert(queue.enqueue(farDeadlineTask) == -1);
    assert(queue.rejectedCount() == 2);

    std::cout << "Test test_policy_filtering_on_enqueue: PASS" << std::endl;
}

// Test bestIndex(), best(), and dequeue() with default priority (no policy)
void test_ranking_and_dequeue_no_policy() {
    TaskQueue queue(5);

    // Empty queue checks
    assert(queue.bestIndex() == -1);
    Task emptyBest = queue.best();
    assert(emptyBest.identifier() == 0);
    Task emptyDequeued = queue.dequeue();
    assert(emptyDequeued.identifier() == 0);
    assert(queue.isEmpty());

    // Without policy, score is determined by Task::weight() = priority * 10 (+25 if overdue)
    Task t1(101, "Low", 2, 10);    // weight: 20
    Task t2(102, "High", 8, 10);   // weight: 80
    Task t3(103, "Mid", 5, 10);    // weight: 50

    queue.enqueue(t1);
    queue.enqueue(t2);
    queue.enqueue(t3);

    assert(queue.bestIndex() == 1);
    assert(queue.best().identifier() == 102);

    Task dequeued1 = queue.dequeue();
    assert(dequeued1.identifier() == 102);
    assert(queue.size() == 2);

    // Remaining: t1 (idx 0, wt 20), t3 (idx 1, wt 50)
    assert(queue.bestIndex() == 1);
    assert(queue.best().identifier() == 103);

    Task dequeued2 = queue.dequeue();
    assert(dequeued2.identifier() == 103);
    assert(queue.size() == 1);

    // Remaining: t1 (idx 0, wt 20)
    assert(queue.bestIndex() == 0);
    Task dequeued3 = queue.dequeue();
    assert(dequeued3.identifier() == 101);
    assert(queue.isEmpty());

    std::cout << "Test test_ranking_and_dequeue_no_policy: PASS" << std::endl;
}

// Test bestIndex(), best(), and dequeue() when policies are applied
void test_ranking_with_policies() {
    TaskQueue queue(5);

    Task t1(1, "First", 5, 20);   // id=1, prio=5, dur=20, ddl=20
    Task t2(2, "Second", 10, 50); // id=2, prio=10, dur=50, ddl=50
    Task t3(3, "Third", 5, 5);    // id=3, prio=5, dur=5, ddl=5

    queue.enqueue(t1);
    queue.enqueue(t2);
    queue.enqueue(t3);

    // 1. FifoPolicy: rank = bias - id * arrivalWeight (lower id = higher rank)
    FifoPolicy fifo(2);
    queue.usePolicy(&fifo);
    // t1 rank = 0 - 1*2 = -2
    // t2 rank = 0 - 2*2 = -4
    // t3 rank = 0 - 3*2 = -6
    assert(queue.bestIndex() == 0);
    assert(queue.best().identifier() == 1);

    // 2. PriorityPolicy: rank = score * 100 - duration + bias
    PriorityPolicy prioPolicy;
    queue.usePolicy(&prioPolicy);
    // t1 rank = 5 * 100 - 20 = 480
    // t2 rank = 10 * 100 - 50 = 950
    // t3 rank = 5 * 100 - 5 = 495
    assert(queue.bestIndex() == 1);
    assert(queue.best().identifier() == 2);

    // 3. DeadlinePolicy: urgency = horizon - deadline; rank = urgency * 5 + bias
    DeadlinePolicy ddlPolicy(100);
    queue.usePolicy(&ddlPolicy);
    // t1 (ddl 20) urgency = 80 -> rank = 400
    // t2 (ddl 50) urgency = 50 -> rank = 250
    // t3 (ddl 5)  urgency = 95 -> rank = 475 (best)
    assert(queue.bestIndex() == 2);
    assert(queue.best().identifier() == 3);

    // Remove best under DeadlinePolicy (t3)
    Task dequeued = queue.dequeue();
    assert(dequeued.identifier() == 3);
    assert(queue.size() == 2);

    // Next best under DeadlinePolicy is t1
    assert(queue.bestIndex() == 0);
    assert(queue.best().identifier() == 1);

    std::cout << "Test test_ranking_with_policies: PASS" << std::endl;
}

// Test totalRemaining() and pendingCount()
void test_aggregate_metrics() {
    TaskQueue queue(10);
    assert(queue.totalRemaining() == 0);
    assert(queue.pendingCount() == 0);

    Task t1(1, "T1", 1, 10);
    Task t2(2, "T2", 1, 20);
    Task t3(3, "T3", 1, 30);

    queue.enqueue(t1);
    queue.enqueue(t2);
    queue.enqueue(t3);

    assert(queue.totalRemaining() == 60);
    assert(queue.pendingCount() == 3);

    // Advance first task partially
    queue.tasks[0].advance(4); // remaining becomes 6, status: started/running (not pending)
    assert(queue.totalRemaining() == 56);
    assert(queue.pendingCount() == 2);

    // Complete first task
    queue.tasks[0].advance(10); // completed
    assert(queue.tasks[0].isComplete());
    assert(queue.totalRemaining() == 50);
    assert(queue.pendingCount() == 2);

    std::cout << "Test test_aggregate_metrics: PASS" << std::endl;
}

// Test advanceAll() and purgeComplete()
void test_advanceAll_and_purgeComplete() {
    TaskQueue queue(5);

    Task t1(1, "Quick", 1, 5);
    Task t2(2, "Medium", 1, 15);
    Task t3(3, "Long", 1, 25);

    queue.enqueue(t1);
    queue.enqueue(t2);
    queue.enqueue(t3);

    // Advance all by 5 units
    // t1: elapsed 5/5 -> complete
    // t2: elapsed 5/15 -> running
    // t3: elapsed 5/25 -> running
    int completedCount = queue.advanceAll(5);
    assert(completedCount == 1);
    assert(queue.tasks[0].isComplete());
    assert(!queue.tasks[1].isComplete());
    assert(!queue.tasks[2].isComplete());
    assert(queue.size() == 3);

    // Purge completed tasks
    int purged = queue.purgeComplete();
    assert(purged == 1);
    assert(queue.size() == 2);
    assert(queue.tasks[0].identifier() == 2);
    assert(queue.tasks[1].identifier() == 3);

    // Advance remaining tasks by 25 units -> both will complete
    completedCount = queue.advanceAll(25);
    assert(completedCount == 2);
    assert(queue.tasks[0].isComplete());
    assert(queue.tasks[1].isComplete());

    purged = queue.purgeComplete();
    assert(purged == 2);
    assert(queue.isEmpty());
    assert(queue.size() == 0);

    // Purging on empty queue
    assert(queue.purgeComplete() == 0);

    std::cout << "Test test_advanceAll_and_purgeComplete: PASS" << std::endl;
}

// Test edge cases: tie-breaking in ranking, zero-duration advance, etc.
void test_edge_cases() {
    TaskQueue queue(5);

    // Tie breaking in bestIndex: first encountered highest rank wins
    Task t1(10, "A", 5, 10);
    Task t2(20, "B", 5, 10);
    queue.enqueue(t1);
    queue.enqueue(t2);

    assert(queue.bestIndex() == 0);
    assert(queue.best().identifier() == 10);

    // Advancing by zero or negative units shouldn't change remaining or status
    int prevRemaining = queue.totalRemaining();
    int finished = queue.advanceAll(0);
    assert(finished == 0);
    assert(queue.totalRemaining() == prevRemaining);
    assert(queue.pendingCount() == 2);

    finished = queue.advanceAll(-5);
    assert(finished == 0);
    assert(queue.totalRemaining() == prevRemaining);
    assert(queue.pendingCount() == 2);

    std::cout << "Test test_edge_cases: PASS" << std::endl;
}

int main() {
    test_constructors_and_initial_state();
    test_enqueue_capacity_and_rejection();
    test_policy_filtering_on_enqueue();
    test_ranking_and_dequeue_no_policy();
    test_ranking_with_policies();
    test_aggregate_metrics();
    test_advanceAll_and_purgeComplete();
    test_edge_cases();

    std::cout << "\nAll TaskQueue tests passed successfully!" << std::endl;
    return 0;
}
```
