Here is the complete C++ unit test suite for testing the `Canvas` class and its methods.

```cpp
#include <iostream>
#include <cassert>
#include <cmath>
#include "Canvas.h"
#include "Shape.h"

// Helper function for floating point comparison
static bool areClose(double a, double b, double epsilon = 1e-6) {
    return std::fabs(a - b) < epsilon;
}

// Custom shape to test polymorphism and proper deletion
class DestructorTrackerShape : public Shape {
public:
    bool* destroyedFlag;
    double shapeArea;

    DestructorTrackerShape(bool* flag, double a = 0.0) : destroyedFlag(flag), shapeArea(a) {
        if (destroyedFlag) *destroyedFlag = false;
    }

    virtual ~DestructorTrackerShape() {
        if (destroyedFlag) *destroyedFlag = true;
    }

    virtual double area() const {
        return shapeArea;
    }
};

void test_Canvas_initialState() {
    Canvas canvas;
    assert(canvas.count() == 0);
    assert(areClose(canvas.totalArea(), 0.0));
    assert(areClose(canvas.largestArea(), 0.0));
    assert(canvas.shapes.empty());

    std::cout << "Test test_Canvas_initialState: PASS" << std::endl;
}

void test_Canvas_addNullShape() {
    Canvas canvas;
    canvas.add(nullptr);
    canvas.add(0);

    assert(canvas.count() == 0);
    assert(canvas.shapes.empty());
    assert(areClose(canvas.totalArea(), 0.0));

    std::cout << "Test test_Canvas_addNullShape: PASS" << std::endl;
}

void test_Canvas_addAndCount() {
    Canvas canvas;
    
    Rectangle* rect1 = new Rectangle(4.0, 5.0);
    Circle* circle1 = new Circle(3.0);
    Shape* baseShape = new Shape();

    canvas.add(rect1);
    assert(canvas.count() == 1);

    canvas.add(circle1);
    assert(canvas.count() == 2);

    canvas.add(baseShape);
    assert(canvas.count() == 3);

    assert(canvas.shapes[0] == rect1);
    assert(canvas.shapes[1] == circle1);
    assert(canvas.shapes[2] == baseShape);

    std::cout << "Test test_Canvas_addAndCount: PASS" << std::endl;
}

void test_Canvas_totalArea() {
    Canvas canvas;
    
    // Rectangle 4x5 => area = 20.0
    canvas.add(new Rectangle(4.0, 5.0));
    assert(areClose(canvas.totalArea(), 20.0));

    // Circle r=2 => area = PI * 4 ≈ 12.566370614359
    canvas.add(new Circle(2.0));
    double expectedArea = 20.0 + (3.14159265358979323846 * 4.0);
    assert(areClose(canvas.totalArea(), expectedArea));

    // Base shape => area = 0.0
    canvas.add(new Shape());
    assert(areClose(canvas.totalArea(), expectedArea));

    std::cout << "Test test_Canvas_totalArea: PASS" << std::endl;
}

void test_Canvas_largestArea() {
    Canvas canvas;
    
    // Empty canvas largestArea should be 0.0
    assert(areClose(canvas.largestArea(), 0.0));

    // Add rectangle with area 15.0
    canvas.add(new Rectangle(3.0, 5.0));
    assert(areClose(canvas.largestArea(), 15.0));

    // Add circle with area ~28.274 (radius = 3)
    canvas.add(new Circle(3.0));
    double circleArea = 3.14159265358979323846 * 9.0;
    assert(areClose(canvas.largestArea(), circleArea));

    // Add smaller rectangle (area 2.0) -> largest area should still be the circle
    canvas.add(new Rectangle(1.0, 2.0));
    assert(areClose(canvas.largestArea(), circleArea));

    // Add larger rectangle (area 50.0) -> new largest area
    canvas.add(new Rectangle(10.0, 5.0));
    assert(areClose(canvas.largestArea(), 50.0));

    std::cout << "Test test_Canvas_largestArea: PASS" << std::endl;
}

void test_Canvas_clear() {
    Canvas canvas;
    bool destroyed1 = false;
    bool destroyed2 = false;

    canvas.add(new DestructorTrackerShape(&destroyed1, 10.0));
    canvas.add(new DestructorTrackerShape(&destroyed2, 20.0));

    assert(canvas.count() == 2);
    assert(!destroyed1);
    assert(!destroyed2);

    canvas.clear();

    assert(canvas.count() == 0);
    assert(canvas.shapes.empty());
    assert(destroyed1);
    assert(destroyed2);
    assert(areClose(canvas.totalArea(), 0.0));
    assert(areClose(canvas.largestArea(), 0.0));

    // Calling clear again on empty canvas should be safe
    canvas.clear();
    assert(canvas.count() == 0);

    std::cout << "Test test_Canvas_clear: PASS" << std::endl;
}

void test_Canvas_destructor() {
    bool destroyed1 = false;
    bool destroyed2 = false;

    {
        Canvas canvas;
        canvas.add(new DestructorTrackerShape(&destroyed1));
        canvas.add(new DestructorTrackerShape(&destroyed2));
        assert(!destroyed1);
        assert(!destroyed2);
    } // Canvas goes out of scope and destructor should trigger clear()

    assert(destroyed1);
    assert(destroyed2);

    std::cout << "Test test_Canvas_destructor: PASS" << std::endl;
}

int main() {
    std::cout << "=== Running Canvas Unit Tests ===" << std::endl;

    test_Canvas_initialState();
    test_Canvas_addNullShape();
    test_Canvas_addAndCount();
    test_Canvas_totalArea();
    test_Canvas_largestArea();
    test_Canvas_clear();
    test_Canvas_destructor();

    std::cout << "=== All Canvas Unit Tests Passed Successfully! ===" << std::endl;
    return 0;
}
```
