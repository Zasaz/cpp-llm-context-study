Here is the complete C++ unit test suite covering `SamplePipeline.cpp`.

```cpp
#include <iostream>
#include <cassert>
#include <cmath>
#include <vector>
#include <string>

#include "SamplePipeline.h"
#include "Channel.h"
#include "Filter.h"
#include "Sample.h"
#include "UnitScale.h"

// Helper comparison for floating point numbers
static bool approxEqual(double a, double b, double epsilon = 1e-6) {
    return std::fabs(a - b) < epsilon;
}

// Concrete mock/derived Filter for testing filter interactions
class OffsetFilter : public Filter {
public:
    double offset;
    double rejectThreshold;

    OffsetFilter(double off = 0.0, double rejectAbove = 1e9)
        : offset(off), rejectThreshold(rejectAbove) {}

    double apply(double value) const {
        return value + offset;
    }

    bool rejects(double value) const {
        return value > rejectThreshold;
    }
};

class GainFilter : public Filter {
public:
    double gain;

    GainFilter(double g = 1.0) : gain(g) {}

    double apply(double value) const {
        return value * gain;
    }

    bool rejects(double /*value*/) const {
        return false;
    }
};

void test_constructors_and_defaults() {
    // Default constructor
    SamplePipeline p1;
    assert(p1.acceptedCount() == 0);
    assert(p1.rejectedCount() == 0);
    assert(p1.minimumQuality == 1);
    assert(p1.filterCount() == 0);
    assert(p1.validSamples() == 0);
    assert(!p1.isSaturated());
    assert(p1.describe() == "empty");

    // Parameterized constructor
    SamplePipeline p2("Voltage", 5);
    assert(p2.channel.channelName() == "Voltage");
    assert(p2.channel.capacity == 5);
    assert(p2.acceptedCount() == 0);
    assert(p2.rejectedCount() == 0);
    assert(p2.minimumQuality == 1);
    assert(p2.filterCount() == 0);
    assert(p2.describe() == "empty");

    std::cout << "Test test_constructors_and_defaults: PASS" << std::endl;
}

void test_filter_management_and_processing() {
    SamplePipeline pipeline("Temp", 10);

    // Adding null filter should be ignored
    assert(pipeline.addFilter(nullptr) == 0);
    assert(pipeline.filterCount() == 0);

    OffsetFilter f1(5.0);   // +5
    GainFilter f2(2.0);     // *2

    assert(pipeline.addFilter(&f1) == 1);
    assert(pipeline.addFilter(&f2) == 2);
    assert(pipeline.filterCount() == 2);

    // Process reading: (10 + 5) * 2 = 30
    double processed = pipeline.process(10.0);
    assert(approxEqual(processed, 30.0));

    // Test stage processing individually
    assert(approxEqual(pipeline.stage(10.0, 0), 15.0)); // f1: 10 + 5
    assert(approxEqual(pipeline.stage(10.0, 1), 20.0)); // f2: 10 * 2

    // Stage out of range bounds
    assert(approxEqual(pipeline.stage(10.0, -1), 10.0));
    assert(approxEqual(pipeline.stage(10.0, 2), 10.0));

    // Test clearFilters
    int cleared = pipeline.clearFilters();
    assert(cleared == 2);
    assert(pipeline.filterCount() == 0);
    assert(approxEqual(pipeline.process(10.0), 10.0));

    std::cout << "Test test_filter_management_and_processing: PASS" << std::endl;
}

void test_would_reject() {
    SamplePipeline pipeline("Pressure", 10);
    OffsetFilter f1(0.0, 100.0); // rejects values > 100.0

    pipeline.addFilter(&f1);
    assert(!pipeline.wouldReject(50.0));
    assert(!pipeline.wouldReject(100.0));
    assert(pipeline.wouldReject(100.1));

    std::cout << "Test test_would_reject: PASS" << std::endl;
}

void test_require_quality() {
    SamplePipeline pipeline("Test", 5);
    assert(pipeline.minimumQuality == 1);

    int prev = pipeline.requireQuality(3);
    assert(prev == 1);
    assert(pipeline.minimumQuality == 3);

    // Negative requirement should not change the quality
    prev = pipeline.requireQuality(-1);
    assert(prev == 3);
    assert(pipeline.minimumQuality == 3);

    std::cout << "Test test_require_quality: PASS" << std::endl;
}

void test_ingest_and_rejections() {
    SamplePipeline pipeline("Current", 2); // capacity = 2
    OffsetFilter f1(2.0, 50.0);           // adds 2, rejects raw > 50
    pipeline.addFilter(&f1);
    pipeline.requireQuality(2);

    // 1. Rejected due to quality < minimumQuality
    int res1 = pipeline.ingest(0, 10.0, 1);
    assert(res1 == -1);
    assert(pipeline.acceptedCount() == 0);
    assert(pipeline.rejectedCount() == 1);

    // 2. Rejected due to filter wouldReject
    int res2 = pipeline.ingest(1, 60.0, 2);
    assert(res2 == -1);
    assert(pipeline.acceptedCount() == 0);
    assert(pipeline.rejectedCount() == 2);

    // 3. Successfully ingested: reading = 10.0 -> filtered = 12.0
    int res3 = pipeline.ingest(2, 10.0, 2);
    assert(res3 == 1);
    assert(pipeline.acceptedCount() == 1);
    assert(pipeline.rejectedCount() == 2);
    assert(approxEqual(pipeline.latest().value, 12.0));

    // 4. Second success: reading = 20.0 -> filtered = 22.0
    int res4 = pipeline.ingest(3, 20.0, 3);
    assert(res4 == 2);
    assert(pipeline.acceptedCount() == 2);
    assert(pipeline.isSaturated());

    // 5. Channel capacity exceeded (saturated)
    int res5 = pipeline.ingest(4, 30.0, 2);
    assert(res5 == -1);
    assert(pipeline.acceptedCount() == 2);
    assert(pipeline.rejectedCount() == 3);

    std::cout << "Test test_ingest_and_rejections: PASS" << std::endl;
}

void test_ingest_batch() {
    SamplePipeline pipeline("BatchChan", 5);
    OffsetFilter f1(1.0, 10.0); // Rejects > 10.0
    pipeline.addFilter(&f1);

    std::vector<double> batch = { 2.0, 5.0, 15.0, 8.0 }; // 15.0 will be rejected
    int taken = pipeline.ingestBatch(batch, 1);

    assert(taken == 3);
    assert(pipeline.acceptedCount() == 3);
    assert(pipeline.rejectedCount() == 1);
    assert(pipeline.validSamples() == 3);

    std::cout << "Test test_ingest_batch: PASS" << std::endl;
}

void test_channel_aggregations_and_scale() {
    SamplePipeline pipeline("Sensor", 10);
    pipeline.scale = UnitScale(2.0, 1.0, "V"); // Example scale configuration

    pipeline.ingest(0, 10.0, 1);
    pipeline.ingest(1, 20.0, 1);
    pipeline.ingest(2, 30.0, 1);

    assert(approxEqual(pipeline.channelMean(), 20.0));
    assert(approxEqual(pipeline.channelPeak(), 30.0));

    // UnitScale conversions
    double norm = pipeline.normalised(10.0);
    double denorm = pipeline.denormalised(norm);
    assert(approxEqual(denorm, 10.0));

    // Describe
    std::string desc = pipeline.describe();
    assert(desc == "Sensor:" + pipeline.scale.unitName());

    std::cout << "Test test_channel_aggregations_and_scale: PASS" << std::endl;
}

void test_flush() {
    SamplePipeline pipeline("Buffer", 5);
    pipeline.ingest(0, 10.0, 1);
    pipeline.ingest(1, 20.0, 1);
    pipeline.ingest(2, 20.0, 0); // rejected due to quality default minimum 1

    assert(pipeline.acceptedCount() == 2);
    assert(pipeline.rejectedCount() == 1);
    assert(!pipeline.channel.isEmpty());

    int flushedCount = pipeline.flush();
    assert(flushedCount == 2);
    assert(pipeline.acceptedCount() == 0);
    assert(pipeline.rejectedCount() == 0);
    assert(pipeline.validSamples() == 0);
    assert(pipeline.channel.isEmpty());
    assert(pipeline.describe() == "empty");
    assert(pipeline.channel.channelName() == "Buffer");
    assert(pipeline.channel.capacity == 5);

    std::cout << "Test test_flush: PASS" << std::endl;
}

int main() {
    test_constructors_and_defaults();
    test_filter_management_and_processing();
    test_would_reject();
    test_require_quality();
    test_ingest_and_rejections();
    test_ingest_batch();
    test_channel_aggregations_and_scale();
    test_flush();

    std::cout << "\nAll SamplePipeline tests passed successfully!" << std::endl;
    return 0;
}
```
