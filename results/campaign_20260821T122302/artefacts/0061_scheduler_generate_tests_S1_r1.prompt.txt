    floorPriority = 0;

    int score = task.priority;
    if (score < floorPriority) {
        score = floorPriority;
    }
    return score * 100 - task.duration + bias;
}

    return "priority";

    if (urgency < 0) {
        urgency = 0;

    return urgency * 5 + bias;

    return "deadline";

#include "ResourceLedger.h"

ResourceLedger::ResourceLedger() {
    limit = 100;
}

ResourceLedger::ResourceLedger(int perResourceLimit) {
    limit = perResourceLimit > 0 ? perResourceLimit : 1;
}

int ResourceLedger::deposit(const std::string& resource, int amount) {
    if (amount <= 0) {
        return balanceOf(resource);
    }
    int current = balanceOf(resource);
    int updated = current + amount;
    if (updated > limit) {
        updated = limit;
    }
    balances[resource] = updated;
    movements.record(updated - current);
    return updated;
}

int ResourceLedger::withdraw(const std::string& resource, int amount) {
    int current = balanceOf(resource);
    if (amount <= 0 || amount > current) {
        return -1;
    }
    balances[resource] = current - amount;
    movements.record(amount);
    return current - amount;
}

int ResourceLedger::balanceOf(const std::string& resource) const {
    std::map<std::string, int>::const_iterator found = balances.find(resource);
    if (found == balances.end()) {
        return 0;
    }
    return found->second;
}

bool ResourceLedger::holds(const std::string& resource) const {
    return balances.find(resource) != balances.end();
}

int ResourceLedger::resourceCount() const {
    return static_cast<int>(balances.size());
}

int ResourceLedger::totalHeld() const {
    int sum = 0;
    std::map<std::string, int>::const_iterator it = balances.begin();
    while (it != balances.end()) {
        sum = sum + it->second;
        ++it;
    }
    return sum;
}

double ResourceLedger::averageMovement() const {
    return movements.mean();
}

int ResourceLedger::clearAll() {
    int previous = resourceCount();
    balances.clear();
    movements.discard();
    return previous;
}

    if (value < 0) {
        ignored = ignored + 1;
        return observations();
    }

    return observations();

int RunStatistics::observations() const {
    return static_cast<int>(samples.size());
}

int RunStatistics::total() const {
    int sum = 0;
    for (unsigned int i = 0; i < samples.size(); ++i) {
        sum = sum + samples[i];
    }
    return sum;
}

double RunStatistics::mean() const {
    int n = observations();
    if (n == 0) {
        return 0.0;
    }
    return static_cast<double>(total()) / static_cast<double>(n);
}

int RunStatistics::minimum() const {
    if (samples.empty()) {
        return 0;
    }
    int best = samples[0];
    for (unsigned int i = 1; i < samples.size(); ++i) {
        if (samples[i] < best) {
            best = samples[i];
        }
    }
    return best;
}

int RunStatistics::maximum() const {
    if (samples.empty()) {
        return 0;
    }
    int best = samples[0];
    for (unsigned int i = 1; i < samples.size(); ++i) {
        if (samples[i] > best) {
            best = samples[i];
        }
    }
    return best;
}

int Scheduler::step(int units) {
    if (units <= 0) {
        return completed;
    }
    tick = tick + units;
    int finished = queue.advanceAll(units);
    completed = completed + finished;
    stats.record(finished);
    queue.purgeComplete();
    return completed;
}

int Scheduler::runUntilIdle(int maxSteps) {
    int taken = 0;
    while (taken < maxSteps && !idle()) {
        step(1);
        taken = taken + 1;
    }
    return taken;
}

    return id;
}

    return name;

int Task::weight() const {
    int base = priority * 10;
    if (isOverdue()) {
        base = base + 25;
    }
    return base;
}

int Task::remaining() const {
    int left = duration - elapsed;
    return left > 0 ? left : 0;
}

bool Task::isPending() const {
    return !started && !finished;
}

bool Task::isRunning() const {
    return started && !finished;
}

bool Task::isComplete() const {
    return finished;
}

bool Task::isOverdue() const {
    return elapsed > deadline;
}

int Task::advance(int units) {
    if (units <= 0 || finished) {
        return remaining();
    }
    started = true;
    elapsed = elapsed + units;
    if (elapsed >= duration) {
        elapsed = duration;
        finished = true;
    }
    return remaining();
}

int Task::restart() {
    int previous = elapsed;
    elapsed = 0;
    started = false;
    finished = false;
    return previous;
}

int Task::setDeadline(int limit) {
    int previous = deadline;
    if (limit >= 0) {
        deadline = limit;
    }
    return previous;
}

#include "TaskQueue.h"

TaskQueue::TaskQueue() {
    policy = 0;
    capacity = 8;
    rejected = 0;
}

TaskQueue::TaskQueue(int limit) {
    policy = 0;
    capacity = limit > 0 ? limit : 1;
    rejected = 0;
}

int TaskQueue::size() const {
    return static_cast<int>(tasks.size());
}

bool TaskQueue::isEmpty() const {
    return tasks.empty();
}

bool TaskQueue::isFull() const {
    return size() >= capacity;
}

int TaskQueue::enqueue(const Task& task) {
    if (isFull()) {
        rejected = rejected + 1;
        return -1;
    }
    if (policy != 0 && !policy->accepts(task)) {
        rejected = rejected + 1;
        return -1;
    }
    tasks.push_back(task);
    return size();
}

int TaskQueue::usePolicy(Policy* replacement) {
    policy = replacement;
    return size();
}

int TaskQueue::bestIndex() const {
    if (tasks.empty()) {
        return -1;
    }
    int chosen = 0;
    int bestScore = 0;
    for (unsigned int i = 0; i < tasks.size(); ++i) {
        int score = 0;
        if (policy != 0) {
            score = policy->rank(tasks[i]);
        } else {
            score = tasks[i].weight();
        }
        if (i == 0 || score > bestScore) {
            chosen = static_cast<int>(i);
            bestScore = score;
        }
    }
    return chosen;
}

Task TaskQueue::best() const {
    int index = bestIndex();
    if (index < 0) {
        return Task();
    }
    return tasks[static_cast<unsigned int>(index)];
}

Task TaskQueue::dequeue() {
    int index = bestIndex();
    if (index < 0) {
        return Task();
    }
    Task chosen = tasks[static_cast<unsigned int>(index)];
    tasks.erase(tasks.begin() + index);
    return chosen;
}

int TaskQueue::totalRemaining() const {
    int total = 0;
    for (unsigned int i = 0; i < tasks.size(); ++i) {
        total = total + tasks[i].remaining();
    }
    return total;
}

int TaskQueue::pendingCount() const {
    int total = 0;
    for (unsigned int i = 0; i < tasks.size(); ++i) {
        if (tasks[i].isPending()) {
            total = total + 1;
        }
    }
    return total;
}

int TaskQueue::advanceAll(int units) {
    int completed = 0;
    for (unsigned int i = 0; i < tasks.size(); ++i) {
        tasks[i].advance(units);
        if (tasks[i].isComplete()) {
            completed = completed + 1;
        }
    }
    return completed;
}

int TaskQueue::purgeComplete() {
    std::vector<Task> kept;
    int removed = 0;
    for (unsigned int i = 0; i < tasks.size(); ++i) {
        if (tasks[i].isComplete()) {
            removed = removed + 1;
        } else {
            kept.push_back(tasks[i]);
        }
    }
    tasks = kept;
    return removed;
}

int TaskQueue::rejectedCount() const {
    return rejected;
}
