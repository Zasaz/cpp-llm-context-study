#include <iostream>
#include <cassert>
#include <cmath>
#include <vector>
#include <string>

#include "SamplePipeline.h"
#include "Channel.h"
#include "Filter.h"
#include "Sample.h"
#include "UnitScale.h"

// Helper function for floating point comparisons
static bool approxEqual(double a, double b, double epsilon = 1e-6) {
    return std::fabs(a - b) <= epsilon;
}

// Test default constructor and parameterized constructor initialization
void test_SamplePipeline_constructors() {
    SamplePipeline defaultPipeline;
    assert(defaultPipeline.acceptedCount() == 0);
    assert(defaultPipeline.rejectedCount() == 0);
    assert(defaultPipeline.filterCount() == 0);
    assert(defaultPipeline.minimumQuality == 1);
    assert(defaultPipeline.channel.channelName() == "unnamed");
    assert(defaultPipeline.channel.capacity == 16);
    assert(defaultPipeline.channel.isEmpty());
    assert(defaultPipeline.describe() == "empty");

    SamplePipeline customPipeline("SensorA", 5);
    assert(customPipeline.acceptedCount() == 0);
    assert(customPipeline.rejectedCount() == 0);
    assert(customPipeline.filterCount() == 0);
    assert(customPipeline.minimumQuality == 1);
    assert(customPipeline.channel.channelName() == "SensorA");
    assert(customPipeline.channel.capacity == 5);
    assert(customPipeline.channel.isEmpty());
    assert(customPipeline.describe() == "empty");

    std::cout << "Test test_SamplePipeline_constructors: PASS" << std::endl;
}

// Test filter management: addFilter, filterCount, and clearFilters
void test_SamplePipeline_filterManagement() {
    SamplePipeline pipeline("TestChannel", 10);

    // Adding null filter should not change count
    assert(pipeline.addFilter(nullptr) == 0);
    assert(pipeline.filterCount() == 0);

    PassFilter passFilter;
    ClampFilter clampFilter(0.0, 10.0);
    SmoothFilter smoothFilter(0.5);

    assert(pipeline.addFilter(&passFilter) == 1);
    assert(pipeline.filterCount() == 1);

    assert(pipeline.addFilter(&clampFilter) == 2);
    assert(pipeline.filterCount() == 2);

    assert(pipeline.addFilter(&smoothFilter) == 3);
    assert(pipeline.filterCount() == 3);

    // Test clearFilters
    int removed = pipeline.clearFilters();
    assert(removed == 3);
    assert(pipeline.filterCount() == 0);

    // Clearing empty list returns 0
    assert(pipeline.clearFilters() == 0);

    std::cout << "Test test_SamplePipeline_filterManagement: PASS" << std::endl;
}

// Test pipeline process() and stage() operations
void test_SamplePipeline_processAndStage() {
    SamplePipeline pipeline("ProcessTest", 10);
    
    // Processing with no filters returns the original reading
    assert(approxEqual(pipeline.process(42.5), 42.5));

    ClampFilter clampFilter(0.0, 50.0);
    SmoothFilter smoothFilter(0.5); // previous starts unprimed
    pipeline.addFilter(&clampFilter);
    pipeline.addFilter(&smoothFilter);

    // stage() out of bounds returns original value
    assert(approxEqual(pipeline.stage(100.0, -1), 100.0));
    assert(approxEqual(pipeline.stage(100.0, 2), 100.0));

    // stage 0 is clampFilter: clamps 100.0 to 50.0
    assert(approxEqual(pipeline.stage(100.0, 0), 50.0));

    // process applies clamp (100 -> 50) and then smooth (first time primed with 50 -> 50)
    pipeline.clearFilters();
    ClampFilter clamp2(10.0, 20.0);
    pipeline.addFilter(&clamp2);
    assert(approxEqual(pipeline.process(5.0), 10.0));
    assert(approxEqual(pipeline.process(15.0), 15.0));
    assert(approxEqual(pipeline.process(25.0), 20.0));

    std::cout << "Test test_SamplePipeline_processAndStage: PASS" << std::endl;
}

// Test wouldReject() logic with multiple filters
void test_SamplePipeline_wouldReject() {
    SamplePipeline pipeline("RejectTest", 10);
    ClampFilter clampFilter(0.0, 10.0);
    pipeline.addFilter(&clampFilter);

    assert(pipeline.wouldReject(5.0) == false);
    assert(pipeline.wouldReject(-0.1) == true);
    assert(pipeline.wouldReject(10.1) == true);

    PassFilter passFilter;
    pipeline.clearFilters();
    pipeline.addFilter(&passFilter);
    assert(pipeline.wouldReject(50.0) == false);

    // Disabled filter rejects non-zero values
    passFilter.toggle(); // enabled becomes false
    assert(pipeline.wouldReject(10.0) == true);
    assert(pipeline.wouldReject(0.0) == false);

    std::cout << "Test test_SamplePipeline_wouldReject: PASS" << std::endl;
}

// Test single ingestion: ingest() success, quality thresholding, rejection, and capacity saturation
void test_SamplePipeline_ingest() {
    SamplePipeline pipeline("IngestChannel", 2);
    pipeline.requireQuality(50);

    ClampFilter clamp(0.0, 100.0);
    pipeline.addFilter(&clamp);

    // Reject due to low quality
    int res1 = pipeline.ingest(0, 50.0, 40);
    assert(res1 == -1);
    assert(pipeline.acceptedCount() == 0);
    assert(pipeline.rejectedCount() == 1);

    // Reject due to filter rejection
    int res2 = pipeline.ingest(1, 150.0, 80); // ClampFilter rejects reading > 100.0
    assert(res2 == -1);
    assert(pipeline.acceptedCount() == 0);
    assert(pipeline.rejectedCount() == 2);

    // Successful ingestion 1
    int res3 = pipeline.ingest(2, 60.0, 90);
    assert(res3 == 1);
    assert(pipeline.acceptedCount() == 1);
    assert(pipeline.rejectedCount() == 2);
    assert(pipeline.latest().value() == 60.0);
    assert(pipeline.latest().position() == 2);

    // Successful ingestion 2 (fills capacity)
    int res4 = pipeline.ingest(3, 70.0, 95);
    assert(res4 == 2);
    assert(pipeline.acceptedCount() == 2);
    assert(pipeline.isSaturated() == true);

    // Ingestion when channel is full should reject
    int res5 = pipeline.ingest(4, 80.0, 90);
    assert(res5 == -1);
    assert(pipeline.acceptedCount() == 2);
    assert(pipeline.rejectedCount() == 3);

    std::cout << "Test test_SamplePipeline_ingest: PASS" << std::endl;
}

// Test batch ingestion: ingestBatch()
void test_SamplePipeline_ingestBatch() {
    SamplePipeline pipeline("BatchChannel", 5);
    ClampFilter clamp(0.0, 10.0);
    pipeline.addFilter(&clamp);

    std::vector<double> batch = { 2.0, 12.0, 4.0, -1.0, 6.0, 8.0 }; // 12.0 and -1.0 rejected by clamp
    int taken = pipeline.ingestBatch(batch, 75);

    // 2.0 (ok), 12.0 (reject), 4.0 (ok), -1.0 (reject), 6.0 (ok), 8.0 (ok) -> 4 accepted
    assert(taken == 4);
    assert(pipeline.acceptedCount() == 4);
    assert(pipeline.rejectedCount() == 2);
    assert(pipeline.validSamples() == 4);
    assert(!pipeline.isSaturated());

    std::cout << "Test test_SamplePipeline_ingestBatch: PASS" << std::endl;
}

// Test quality requirement manipulation
void test_SamplePipeline_requireQuality() {
    SamplePipeline pipeline;
    assert(pipeline.minimumQuality == 1);

    int prev = pipeline.requireQuality(70);
    assert(prev == 1);
    assert(pipeline.minimumQuality == 70);

    // Negative minimum should be ignored
    prev = pipeline.requireQuality(-5);
    assert(prev == 70);
    assert(pipeline.minimumQuality == 70);

    std::cout << "Test test_SamplePipeline_requireQuality: PASS" << std::endl;
}

// Test channel metrics forwarded by pipeline: channelMean, channelPeak, validSamples
void test_SamplePipeline_channelMetrics() {
    SamplePipeline pipeline("MetricsChannel", 10);
    pipeline.requireQuality(1);

    assert(pipeline.channelMean() == 0.0);
    assert(pipeline.channelPeak() == 0.0);
    assert(pipeline.validSamples() == 0);

    pipeline.ingest(0, 10.0, 80);
    pipeline.ingest(1, 20.0, 90);
    pipeline.ingest(2, 30.0, 70);

    assert(approxEqual(pipeline.channelMean(), 20.0));
    assert(approxEqual(pipeline.channelPeak(), 30.0));
    assert(pipeline.validSamples() == 3);

    std::cout << "Test test_SamplePipeline_channelMetrics: PASS" << std::endl;
}

// Test scaling conversions: normalised and denormalised
void test_SamplePipeline_scaling() {
    SamplePipeline pipeline("ScaleChannel", 10);
    pipeline.scale = UnitScale(2.0, 5.0, "volt");

    // denormalised = reading * 2.0 + 5.0
    assert(approxEqual(pipeline.denormalised(10.0), 25.0));

    // normalised = (reading - 5.0) / 2.0
    assert(approxEqual(pipeline.normalised(25.0), 10.0));

    std::cout << "Test test_SamplePipeline_scaling: PASS" << std::endl;
}

// Test pipeline flush and describe functions
void test_SamplePipeline_flushAndDescribe() {
    SamplePipeline pipeline("TempSensor", 5);
    pipeline.scale = UnitScale(1.0, 0.0, "degC");

    assert(pipeline.describe() == "empty");

    pipeline.ingest(0, 25.5, 100);
    assert(pipeline.describe() == "TempSensor:degC");
    assert(pipeline.acceptedCount() == 1);

    int flushedCount = pipeline.flush();
    assert(flushedCount == 1);
    assert(pipeline.acceptedCount() == 0);
    assert(pipeline.rejectedCount() == 0);
    assert(pipeline.channel.isEmpty());
    assert(pipeline.describe() == "empty");

    std::cout << "Test test_SamplePipeline_flushAndDescribe: PASS" << std::endl;
}

    test_SamplePipeline_processAndStage();
    test_SamplePipeline_wouldReject();
    test_SamplePipeline_ingest();
    test_SamplePipeline_ingestBatch();
    test_SamplePipeline_requireQuality();
    test_SamplePipeline_channelMetrics();
    test_SamplePipeline_scaling();
    test_SamplePipeline_flushAndDescribe();

    std::cout << "All SamplePipeline unit tests passed successfully!" << std::endl;
    return 0;
}
