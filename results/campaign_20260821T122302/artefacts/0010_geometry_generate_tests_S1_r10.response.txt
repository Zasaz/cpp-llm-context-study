#include <iostream>
#include <cassert>
#include <cmath>
#include "Canvas.h"
#include "Shape.h"

// Helper function for floating-point comparisons
static bool approxEqual(double a, double b, double epsilon = 1e-6) {
    return std::fabs(a - b) <= epsilon;
}

void test_canvas_initial_state() {
    Canvas canvas;
    assert(canvas.count() == 0);
    assert(canvas.shapes.empty());
    assert(approxEqual(canvas.totalArea(), 0.0));
    assert(approxEqual(canvas.largestArea(), 0.0));

    std::cout << "Test test_canvas_initial_state: PASS" << std::endl;
}

void test_canvas_add_null() {
    Canvas canvas;
    canvas.add(nullptr);
    assert(canvas.count() == 0);
    assert(canvas.shapes.empty());
    assert(approxEqual(canvas.totalArea(), 0.0));
    assert(approxEqual(canvas.largestArea(), 0.0));

    std::cout << "Test test_canvas_add_null: PASS" << std::endl;
}

void test_canvas_add_and_count() {
    Canvas canvas;

    Shape* rect1 = new Rectangle(4.0, 5.0);
    Shape* circ1 = new Circle(3.0);

    canvas.add(rect1);
    assert(canvas.count() == 1);
    assert(canvas.shapes.size() == 1);
    assert(canvas.shapes[0] == rect1);

    canvas.add(circ1);
    assert(canvas.count() == 2);
    assert(canvas.shapes.size() == 2);
    assert(canvas.shapes[1] == circ1);

    // canvas destructor will deallocate the shapes
    std::cout << "Test test_canvas_add_and_count: PASS" << std::endl;
}

void test_canvas_total_area() {
    Canvas canvas;

    // Rectangle of 4 x 5 = 20.0
    // Circle of radius 2.0 = pi * 4 = 12.566370614359...
    // Base Shape area = 0.0
    canvas.add(new Rectangle(4.0, 5.0));
    canvas.add(new Circle(2.0));
    canvas.add(new Shape());

    double expectedTotal = (4.0 * 5.0) + (3.14159265358979323846 * 2.0 * 2.0) + 0.0;
    assert(approxEqual(canvas.totalArea(), expectedTotal));

    std::cout << "Test test_canvas_total_area: PASS" << std::endl;
}

void test_canvas_largest_area() {
    Canvas canvas;

    // Add shapes with varying areas
    canvas.add(new Rectangle(2.0, 3.0));    // area = 6.0
    canvas.add(new Rectangle(10.0, 10.0));  // area = 100.0 (largest)
    canvas.add(new Circle(4.0));            // area = pi * 16 ~ 50.265
    canvas.add(new Rectangle(1.0, 1.0));    // area = 1.0

    assert(approxEqual(canvas.largestArea(), 100.0));

    std::cout << "Test test_canvas_largest_area: PASS" << std::endl;
}

void test_canvas_clear() {
    Canvas canvas;

    canvas.add(new Rectangle(3.0, 6.0));
    canvas.add(new Circle(1.5));
    assert(canvas.count() == 2);

    canvas.clear();

    assert(canvas.count() == 0);
    assert(canvas.shapes.empty());
    assert(approxEqual(canvas.totalArea(), 0.0));
    assert(approxEqual(canvas.largestArea(), 0.0));

    // Calling clear again on an empty canvas should be safe
    canvas.clear();
    assert(canvas.count() == 0);

    std::cout << "Test test_canvas_clear: PASS" << std::endl;
}

void test_canvas_reuse_after_clear() {
    Canvas canvas;

    canvas.add(new Rectangle(2.0, 2.0)); // area = 4.0
    assert(canvas.count() == 1);
    assert(approxEqual(canvas.totalArea(), 4.0));

    canvas.clear();
    assert(canvas.count() == 0);

    // Re-add shapes after clear
    canvas.add(new Rectangle(5.0, 6.0)); // area = 30.0
    canvas.add(new Circle(1.0));         // area = pi * 1.0 = ~3.14159

    assert(canvas.count() == 2);
    double expectedTotal = 30.0 + 3.14159265358979323846;
    assert(approxEqual(canvas.totalArea(), expectedTotal));
    assert(approxEqual(canvas.largestArea(), 30.0));

    std::cout << "Test test_canvas_reuse_after_clear: PASS" << std::endl;
}

void test_canvas_destructor_cleanup() {
    // Test that a Canvas going out of scope safely frees its allocated shapes
    {
        Canvas localCanvas;
        localCanvas.add(new Rectangle(10.0, 20.0));
        localCanvas.add(new Circle(5.0));
        assert(localCanvas.count() == 2);
    } // destructor invoked here without crashes or leaks

    std::cout << "Test test_canvas_destructor_cleanup: PASS" << std::endl;
}

int main() {
    test_canvas_initial_state();
    test_canvas_add_null();
    test_canvas_add_and_count();
    test_canvas_total_area();
    test_canvas_largest_area();
    test_canvas_clear();
    test_canvas_reuse_after_clear();
    test_canvas_destructor_cleanup();

    std::cout << "All Canvas tests passed successfully!" << std::endl;
    return 0;
}
