Item::Item() {

    name = itemName;
    quantity = itemQuantity;
    unitPrice = price;

}

bool Item::isInStock() const {
    return quantity > 0;

    if (amount > 0) {
        quantity = quantity + amount;

}

bool Item::consume(int amount) {
    if (amount <= 0 || amount > quantity) {
        return false;
    }
    quantity = quantity - amount;
    return true;
}

        return false;
