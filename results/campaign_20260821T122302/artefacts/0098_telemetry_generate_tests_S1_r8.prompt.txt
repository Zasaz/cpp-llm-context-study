#include "Channel.h"

Channel::Channel() {
    name = "unnamed";
    capacity = 16;
    dropped = 0;
}

Channel::Channel(const std::string& channelName, int limit) {
    name = channelName;
    capacity = limit > 0 ? limit : 1;
    dropped = 0;
}

int Channel::append(const Sample& sample) {
    if (isFull()) {
        dropped = dropped + 1;
        return -1;
    }
    samples.push_back(sample);
    return count();
}

int Channel::count() const {
    return static_cast<int>(samples.size());
}

int Channel::validCount() const {
    int total = 0;
    for (unsigned int i = 0; i < samples.size(); ++i) {
        if (samples[i].isValid()) {
            total = total + 1;
        }
    }
    return total;
}

int Channel::droppedCount() const {
    return dropped;
}

bool Channel::isFull() const {
    return count() >= capacity;
}

bool Channel::isEmpty() const {
    return samples.empty();
}

double Channel::sum() const {
    double total = 0.0;
    for (unsigned int i = 0; i < samples.size(); ++i) {
        total = total + samples[i].value();
    }
    return total;
}

double Channel::mean() const {
    int n = count();
    if (n == 0) {
        return 0.0;
    }
    return sum() / static_cast<double>(n);
}

double Channel::peak() const {
    if (samples.empty()) {
        return 0.0;
    }
    double best = samples[0].value();
    for (unsigned int i = 1; i < samples.size(); ++i) {
        double current = samples[i].value();
        if (current > best) {
            best = current;
        }
    }
    return best;
}

double Channel::trough() const {
    if (samples.empty()) {
        return 0.0;
    }
    double best = samples[0].value();
    for (unsigned int i = 1; i < samples.size(); ++i) {
        double current = samples[i].value();
        if (current < best) {
            best = current;
        }
    }
    return best;
}

double Channel::range() const {
    return peak() - trough();
}

double Channel::meanInBase() const {
    return scale.toBase(mean());
}

Sample Channel::sampleAt(int position) const {
    if (position < 0 || position >= count()) {
        return Sample();
    }
    return samples[static_cast<unsigned int>(position)];
}

Sample Channel::newest() const {
    return sampleAt(count() - 1);
}

int Channel::purgeDegraded() {
    std::vector<Sample> kept;
    int removed = 0;
    for (unsigned int i = 0; i < samples.size(); ++i) {
        if (samples[i].isDegraded()) {
            removed = removed + 1;
        } else {
            kept.push_back(samples[i]);
        }
    }
    samples = kept;
    return removed;
}

std::string Channel::channelName() const {
    return name;
}

int Channel::rescale(const UnitScale& replacement) {
    scale = replacement;
    return count();
}

int Dashboard::sweep() {
    int raised = 0;
    for (unsigned int i = 0; i < panels.size(); ++i) {
        double reading = panels[i].mean();
        for (unsigned int r = 0; r < rules.size(); ++r) {
            int before = rules[r]->firedCount();
            if (rules[r]->evaluate(reading) > before) {
                raised = raised + 1;
            }
        }
    }
    checks = checks + 1;
    return raised;
}

int Dashboard::alertsFor(double reading) const {
    int total = 0;
    for (unsigned int r = 0; r < rules.size(); ++r) {
        if (rules[r]->triggers(reading)) {
            total = total + 1;
        }
    }
    return total;
}

int Dashboard::worstSeverity() const {
    int worst = 0;
    for (unsigned int r = 0; r < rules.size(); ++r) {
        int current = rules[r]->severity();
        if (current > worst) {
            worst = current;
        }
    }
    return worst;
}

double Dashboard::panelMean(int index) const {
    if (index < 0 || index >= panelCount()) {
        return 0.0;
    }
    return panels[static_cast<unsigned int>(index)].mean();
}

double Dashboard::overallMean() const {
    if (panels.empty()) {
        return 0.0;
    }
    double total = 0.0;
    for (unsigned int i = 0; i < panels.size(); ++i) {
        total = total + panels[i].mean();
    }
    return total / static_cast<double>(panels.size());
}

std::string Dashboard::heading() const {
    return title;
}

std::string Dashboard::status() const {
    if (rules.empty()) {
        return "idle";
    }
    if (worstSeverity() >= 3) {
        return "critical";
    }
    return "nominal";
}

    checks = 0;
    return previous;
}

#include "RingBuffer.h"

RingBuffer::RingBuffer() {
    cells.resize(8, 0.0);
    head = 0;
    filled = 0;
    overwrites = 0;
}

RingBuffer::RingBuffer(int slotCount) {
    int size = slotCount > 0 ? slotCount : 1;
    cells.resize(static_cast<unsigned int>(size), 0.0);
    head = 0;
    filled = 0;
    overwrites = 0;
}

int RingBuffer::push(double value) {
    if (isFull()) {
        overwrites = overwrites + 1;
    }
    cells[static_cast<unsigned int>(head)] = value;
    head = (head + 1) % slots();
    if (filled < slots()) {
        filled = filled + 1;
    }
    return filled;
}

double RingBuffer::oldest() const {
    if (isEmpty()) {
        return 0.0;
    }
    int index = (head - filled + slots()) % slots();
    return cells[static_cast<unsigned int>(index)];
}

double RingBuffer::newest() const {
    if (isEmpty()) {
        return 0.0;
    }
    int index = (head - 1 + slots()) % slots();
    return cells[static_cast<unsigned int>(index)];
}

double RingBuffer::at(int offset) const {
    if (offset < 0 || offset >= filled) {
        return 0.0;
    }
    int index = (head - filled + offset + slots()) % slots();
    return cells[static_cast<unsigned int>(index)];
}

int RingBuffer::occupancy() const {
    return filled;
}

int RingBuffer::slots() const {
    return static_cast<int>(cells.size());
}

bool RingBuffer::isFull() const {
    return filled >= slots();
}

bool RingBuffer::isEmpty() const {
    return filled == 0;
}

int RingBuffer::overwriteCount() const {
    return overwrites;
}

double RingBuffer::total() const {
    double sum = 0.0;
    for (int i = 0; i < filled; ++i) {
        sum = sum + at(i);
    }
    return sum;
}

double RingBuffer::average() const {
    if (filled == 0) {
        return 0.0;
    }
    return total() / static_cast<double>(filled);
}

int RingBuffer::drain() {
    int previous = filled;
    head = 0;
    filled = 0;
    for (unsigned int i = 0; i < cells.size(); ++i) {
        cells[i] = 0.0;
    }

int SamplePipeline::addFilter(Filter* filter) {
    if (filter == 0) {
        return filterCount();
    }
    filters.push_back(filter);
    return filterCount();
}

int SamplePipeline::filterCount() const {
    return static_cast<int>(filters.size());
}

int SamplePipeline::clearFilters() {
    int previous = filterCount();
    filters.clear();
    return previous;
}

double SamplePipeline::process(double reading) {
    double current = reading;
    for (unsigned int i = 0; i < filters.size(); ++i) {
        current = filters[i]->apply(current);
    }
    return current;
}

double SamplePipeline::stage(double reading, int index) {
    if (index < 0 || index >= filterCount()) {
        return reading;
    }
    return filters[static_cast<unsigned int>(index)]->apply(reading);
}

bool SamplePipeline::wouldReject(double reading) const {
    for (unsigned int i = 0; i < filters.size(); ++i) {
        if (filters[i]->rejects(reading)) {
            return true;
        }
    }
    return false;
}

int SamplePipeline::ingest(int sequence, double reading, int quality) {
    if (quality < minimumQuality || wouldReject(reading)) {
        rejected = rejected + 1;
        return -1;
    }
    double filtered = process(reading);
    Sample sample(sequence, filtered, quality);
    if (!sample.isValid()) {
        rejected = rejected + 1;
        return -1;
    }
    if (channel.append(sample) < 0) {
        rejected = rejected + 1;
        return -1;
    }
    accepted = accepted + 1;
    return accepted;
}

int SamplePipeline::ingestBatch(const std::vector<double>& readings, int quality) {
    int taken = 0;
    for (unsigned int i = 0; i < readings.size(); ++i) {
        if (ingest(static_cast<int>(i), readings[i], quality) > 0) {
            taken = taken + 1;
        }
    }
    return taken;
}

#include "UnitScale.h"
#include <cmath>

UnitScale::UnitScale() {
    factor = 1.0;
    offset = 0.0;
    unit = "raw";
}

UnitScale::UnitScale(double scaleFactor, double scaleOffset, const std::string& unitName) {
    factor = scaleFactor != 0.0 ? scaleFactor : 1.0;
    offset = scaleOffset;
    unit = unitName;
}

double UnitScale::toBase(double reading) const {
    return reading * factor + offset;
}

double UnitScale::fromBase(double reading) const {
    return (reading - offset) / factor;
}

double UnitScale::spanOf(double low, double high) const {
    double a = toBase(low);
    double b = toBase(high);
    double delta = b - a;
    return delta < 0.0 ? -delta : delta;
}

std::string UnitScale::unitName() const {
    return unit;
}

bool UnitScale::isIdentity() const {
    return factor == 1.0 && offset == 0.0;
}

double UnitScale::quantised(double reading, int digits) const {
    if (digits < 0) {
        return reading;
    }
    double step = std::pow(10.0, static_cast<double>(digits));
    double shifted = reading * step;
    if (shifted < 0.0) {
        return -std::floor(-shifted + 0.5) / step;
    }
    return std::floor(shifted + 0.5) / step;
}

UnitScale UnitScale::inverted() const {
    return UnitScale(1.0 / factor, -offset / factor, unit);
}
