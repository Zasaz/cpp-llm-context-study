Here is the complete C++ unit test suite covering the `TaskQueue` class and its interactions.

```cpp
#include <cassert>
#include <iostream>
#include <vector>
#include <string>

#include "TaskQueue.h"
#include "Policy.h"
#include "Task.h"

// Test default and parameterized constructors, and initial state
void test_constructors_and_initial_state() {
    TaskQueue defaultQueue;
    assert(defaultQueue.size() == 0);
    assert(defaultQueue.isEmpty() == true);
    assert(defaultQueue.isFull() == false);
    assert(defaultQueue.capacity == 8);
    assert(defaultQueue.rejectedCount() == 0);
    assert(defaultQueue.policy == 0);

    TaskQueue customQueue(4);
    assert(customQueue.size() == 0);
    assert(customQueue.capacity == 4);

    // Negative/Zero capacity should fallback to 1
    TaskQueue zeroQueue(0);
    assert(zeroQueue.capacity == 1);

    TaskQueue negativeQueue(-5);
    assert(negativeQueue.capacity == 1);

    std::cout << "Test test_constructors_and_initial_state: PASS" << std::endl;
}

// Test basic enqueue and dequeue operations without any custom policy (default ranking by weight)
void test_enqueue_and_dequeue_default_policy() {
    TaskQueue queue(5);
    Task t1(1, "Task1", 2, 10); // weight: 2 * 10 = 20
    Task t2(2, "Task2", 5, 10); // weight: 5 * 10 = 50
    Task t3(3, "Task3", 1, 10); // weight: 1 * 10 = 10

    assert(queue.enqueue(t1) == 1);
    assert(queue.enqueue(t2) == 2);
    assert(queue.enqueue(t3) == 3);
    assert(queue.size() == 3);
    assert(!queue.isEmpty());
    assert(!queue.isFull());

    // Best index should be t2 (index 1) because weight 50 is the highest
    assert(queue.bestIndex() == 1);
    assert(queue.best().identifier() == 2);

    // Dequeue should return t2 and remove it
    Task popped1 = queue.dequeue();
    assert(popped1.identifier() == 2);
    assert(queue.size() == 2);

    // Next best should be t1 (weight 20 vs weight 10)
    assert(queue.bestIndex() == 0);
    Task popped2 = queue.dequeue();
    assert(popped2.identifier() == 1);
    assert(queue.size() == 1);

    // Last best is t3
    Task popped3 = queue.dequeue();
    assert(popped3.identifier() == 3);
    assert(queue.size() == 0);
    assert(queue.isEmpty());

    // Dequeue from empty queue
    Task emptyPopped = queue.dequeue();
    assert(emptyPopped.identifier() == 0); // default Task
    assert(queue.bestIndex() == -1);
    assert(queue.best().identifier() == 0);

    std::cout << "Test test_enqueue_and_dequeue_default_policy: PASS" << std::endl;
}

// Test queue capacity limits and rejection tracking
void test_capacity_and_overflow() {
    TaskQueue queue(2);
    Task t1(1, "Task1", 1, 5);
    Task t2(2, "Task2", 2, 5);
    Task t3(3, "Task3", 3, 5);

    assert(queue.enqueue(t1) == 1);
    assert(queue.enqueue(t2) == 2);
    assert(queue.isFull());

    // Attempting to enqueue when full
    int res = queue.enqueue(t3);
    assert(res == -1);
    assert(queue.rejectedCount() == 1);
    assert(queue.size() == 2);

    // Another overflow attempt
    assert(queue.enqueue(t3) == -1);
    assert(queue.rejectedCount() == 2);

    // Make space and enqueue again
    queue.dequeue();
    assert(!queue.isFull());
    assert(queue.enqueue(t3) == 2);
    assert(queue.rejectedCount() == 2);

    std::cout << "Test test_capacity_and_overflow: PASS" << std::endl;
}

// Test policy assignment and behavior with PriorityPolicy and FifoPolicy
void test_policy_interaction_and_filtering() {
    TaskQueue queue(5);
    PriorityPolicy priorityPolicy(5); // Only accepts priority >= 5

    queue.usePolicy(&priorityPolicy);

    Task lowPriority(1, "Low", 3, 10);
    Task highPriority1(2, "High1", 6, 20);
    Task highPriority2(3, "High2", 8, 10);

    // lowPriority should be rejected by policy
    assert(queue.enqueue(lowPriority) == -1);
    assert(queue.rejectedCount() == 1);
    assert(queue.size() == 0);

    assert(queue.enqueue(highPriority1) == 1);
    assert(queue.enqueue(highPriority2) == 2);

    // PriorityPolicy ranking: score * 100 - duration + bias
    // highPriority1: 6 * 100 - 20 = 580
    // highPriority2: 8 * 100 - 10 = 790
    assert(queue.bestIndex() == 1);
    assert(queue.best().identifier() == 3);

    // Change policy to FifoPolicy
    FifoPolicy fifoPolicy(2);
    queue.usePolicy(&fifoPolicy);
    // FifoPolicy rank: bias - id * arrivalWeight
    // highPriority1 (id 2): -2 * 2 = -4
    // highPriority2 (id 3): -3 * 2 = -6
    // -4 > -6, so highPriority1 should be chosen first
    assert(queue.bestIndex() == 0);
    assert(queue.best().identifier() == 2);

    std::cout << "Test test_policy_interaction_and_filtering: PASS" << std::endl;
}

// Test DeadlinePolicy filtering and ranking
void test_deadline_policy() {
    TaskQueue queue(5);
    DeadlinePolicy deadlinePolicy(50); // horizon = 50
    queue.usePolicy(&deadlinePolicy);

    Task t1(1, "Task1", 1, 30); // default deadline = duration = 30 <= 50 -> accepted
    Task t2(2, "Task2", 1, 60); // default deadline = 60 > 50 -> rejected

    assert(queue.enqueue(t1) == 1);
    assert(queue.enqueue(t2) == -1);
    assert(queue.rejectedCount() == 1);
    assert(queue.size() == 1);

    Task t3(3, "Task3", 1, 10); // deadline = 10 <= 50 -> accepted
    assert(queue.enqueue(t3) == 2);

    // DeadlinePolicy rank: (horizon - deadline) * 5 + bias
    // t1 (deadline 30): (50 - 30) * 5 = 100
    // t3 (deadline 10): (50 - 10) * 5 = 200
    assert(queue.bestIndex() == 1);
    assert(queue.best().identifier() == 3);

    std::cout << "Test test_deadline_policy: PASS" << std::endl;
}

// Test totalRemaining and pendingCount calculation
void test_total_remaining_and_pending_count() {
    TaskQueue queue(5);
    Task t1(1, "T1", 1, 10);
    Task t2(2, "T2", 1, 20);
    Task t3(3, "T3", 1, 15);

    queue.enqueue(t1);
    queue.enqueue(t2);
    queue.enqueue(t3);

    assert(queue.totalRemaining() == 45);
    assert(queue.pendingCount() == 3);

    // Advance first task partially
    queue.tasks[0].advance(4); // remaining 6, started = true
    assert(queue.totalRemaining() == 41);
    assert(queue.pendingCount() == 2);

    // Advance first task to completion
    queue.tasks[0].advance(10); // completed
    assert(queue.totalRemaining() == 35);
    assert(queue.pendingCount() == 2);

    std::cout << "Test test_total_remaining_and_pending_count: PASS" << std::endl;
}

// Test advanceAll and purgeComplete functionality
void test_advance_all_and_purge_complete() {
    TaskQueue queue(5);
    Task t1(1, "T1", 1, 5);
    Task t2(2, "T2", 1, 10);
    Task t3(3, "T3", 1, 15);

    queue.enqueue(t1);
    queue.enqueue(t2);
    queue.enqueue(t3);

    // Advance all by 5 units: t1 completes (5->0), t2 has 5 left, t3 has 10 left
    int completedCount = queue.advanceAll(5);
    assert(completedCount == 1);
    assert(queue.size() == 3);
    assert(queue.tasks[0].isComplete());
    assert(!queue.tasks[1].isComplete());
    assert(!queue.tasks[2].isComplete());
    assert(queue.totalRemaining() == 15);

    // Purge completed tasks
    int purged = queue.purgeComplete();
    assert(purged == 1);
    assert(queue.size() == 2);
    assert(queue.tasks[0].identifier() == 2);
    assert(queue.tasks[1].identifier() == 3);

    // Advance all by 10 units: both t2 and t3 should complete
    completedCount = queue.advanceAll(10);
    assert(completedCount == 2);
    assert(queue.totalRemaining() == 0);

    purged = queue.purgeComplete();
    assert(purged == 2);
    assert(queue.isEmpty());

    // Purging on empty queue
    assert(queue.purgeComplete() == 0);

    std::cout << "Test test_advance_all_and_purge_complete: PASS" << std::endl;
}

int main() {
    std::cout << "=== Running TaskQueue Unit Tests ===" << std::endl;

    test_constructors_and_initial_state();
    test_enqueue_and_dequeue_default_policy();
    test_capacity_and_overflow();
    test_policy_interaction_and_filtering();
    test_deadline_policy();
    test_total_remaining_and_pending_count();
    test_advance_all_and_purge_complete();

    std::cout << "=== All TaskQueue Unit Tests Passed Successfully! ===" << std::endl;
    return 0;
}
```
