```cpp
#include <iostream>
#include <cassert>
#include <cmath>
#include "Canvas.h"
#include "Shape.h"
#include "Vector2.h"

// Helper function for floating point comparisons
static bool isClose(double a, double b, double epsilon = 1e-6) {
    return std::fabs(a - b) < epsilon;
}

void test_canvas_initialState() {
    Canvas canvas;

    assert(canvas.count() == 0);
    assert(isClose(canvas.totalArea(), 0.0));
    assert(isClose(canvas.largestArea(), 0.0));
    assert(canvas.shapes.empty());

    std::cout << "Test test_canvas_initialState: PASS" << std::endl;
}

void test_canvas_addNull() {
    Canvas canvas;

    canvas.add(nullptr);
    canvas.add(0);

    assert(canvas.count() == 0);
    assert(canvas.shapes.empty());
    assert(isClose(canvas.totalArea(), 0.0));
    assert(isClose(canvas.largestArea(), 0.0));

    std::cout << "Test test_canvas_addNull: PASS" << std::endl;
}

void test_canvas_addAndCount() {
    Canvas canvas;

    canvas.add(new Rectangle(4.0, 5.0));
    assert(canvas.count() == 1);

    canvas.add(new Circle(2.0));
    assert(canvas.count() == 2);

    canvas.add(new Shape());
    assert(canvas.count() == 3);

    // Null addition should not change count
    canvas.add(nullptr);
    assert(canvas.count() == 3);

    std::cout << "Test test_canvas_addAndCount: PASS" << std::endl;
}

void test_canvas_totalArea() {
    Canvas canvas;

    // Rectangle: 4.0 * 5.0 = 20.0
    canvas.add(new Rectangle(4.0, 5.0));
    assert(isClose(canvas.totalArea(), 20.0));

    // Circle: PI * 3.0^2 = 9 * PI ~= 28.2743338823
    const double PI = 3.14159265358979323846;
    canvas.add(new Circle(3.0));
    assert(isClose(canvas.totalArea(), 20.0 + (PI * 9.0)));

    // Base Shape: area is 0.0
    canvas.add(new Shape());
    assert(isClose(canvas.totalArea(), 20.0 + (PI * 9.0)));

    std::cout << "Test test_canvas_totalArea: PASS" << std::endl;
}

void test_canvas_largestArea() {
    Canvas canvas;

    // Empty canvas largestArea should be 0.0
    assert(isClose(canvas.largestArea(), 0.0));

    // Add smaller rectangle (area = 10.0)
    canvas.add(new Rectangle(2.0, 5.0));
    assert(isClose(canvas.largestArea(), 10.0));

    // Add largest rectangle (area = 50.0)
    canvas.add(new Rectangle(10.0, 5.0));
    assert(isClose(canvas.largestArea(), 50.0));

    // Add medium circle (area = PI * 2^2 ~= 12.566)
    canvas.add(new Circle(2.0));
    assert(isClose(canvas.largestArea(), 50.0));

    // Add zero-area base shape
    canvas.add(new Shape());
    assert(isClose(canvas.largestArea(), 50.0));

    std::cout << "Test test_canvas_largestArea: PASS" << std::endl;
}

void test_canvas_clear() {
    Canvas canvas;

    canvas.add(new Rectangle(2.0, 3.0));
    canvas.add(new Circle(1.5));
    canvas.add(new Rectangle(10.0, 10.0));

    assert(canvas.count() == 3);
    assert(canvas.totalArea() > 0.0);

    canvas.clear();

    assert(canvas.count() == 0);
    assert(canvas.shapes.empty());
    assert(isClose(canvas.totalArea(), 0.0));
    assert(isClose(canvas.largestArea(), 0.0));

    // Multiple clears should be safe and idempotent
    canvas.clear();
    assert(canvas.count() == 0);
    assert(canvas.shapes.empty());

    std::cout << "Test test_canvas_clear: PASS" << std::endl;
}

void test_canvas_readdAfterClear() {
    Canvas canvas;

    canvas.add(new Rectangle(2.0, 3.0));
    canvas.clear();
    assert(canvas.count() == 0);

    canvas.add(new Rectangle(5.0, 6.0));
    assert(canvas.count() == 1);
    assert(isClose(canvas.totalArea(), 30.0));
    assert(isClose(canvas.largestArea(), 30.0));

    std::cout << "Test test_canvas_readdAfterClear: PASS" << std::endl;
}

void test_canvas_destructorScope() {
    // Verify scoped destruction does not crash or corrupt memory
    {
        Canvas scopedCanvas;
        scopedCanvas.add(new Rectangle(1.0, 2.0));
        scopedCanvas.add(new Circle(3.0));
        scopedCanvas.add(new Shape());
        assert(scopedCanvas.count() == 3);
    } // scopedCanvas goes out of scope and calls ~Canvas(), deleting all allocated shapes

    std::cout << "Test test_canvas_destructorScope: PASS" << std::endl;
}

int main() {
    std::cout << "=== Running Canvas Unit Tests ===" << std::endl;

    test_canvas_initialState();
    test_canvas_addNull();
    test_canvas_addAndCount();
    test_canvas_totalArea();
    test_canvas_largestArea();
    test_canvas_clear();
    test_canvas_readdAfterClear();
    test_canvas_destructorScope();

    std::cout << "=== All Canvas tests passed successfully! ===" << std::endl;
    return 0;
}
```
