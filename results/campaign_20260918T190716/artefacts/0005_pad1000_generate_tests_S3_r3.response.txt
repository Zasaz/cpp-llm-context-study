Here is the complete C++ unit test suite for testing `Canvas.cpp`.

```cpp
#include <iostream>
#include <cassert>
#include <cmath>
#include "Canvas.h"
#include "Shape.h"

// Helper function to compare floating point numbers
bool isClose(double a, double b, double epsilon = 1e-6) {
    return std::fabs(a - b) < epsilon;
}

// Test initial/empty state of Canvas
void test_Canvas_initialState() {
    Canvas canvas;
    assert(canvas.count() == 0);
    assert(canvas.shapes.empty());
    assert(isClose(canvas.totalArea(), 0.0));
    assert(isClose(canvas.largestArea(), 0.0));

    std::cout << "Test test_Canvas_initialState: PASS" << std::endl;
}

// Test adding null pointer
void test_Canvas_addNull() {
    Canvas canvas;
    canvas.add(nullptr);
    assert(canvas.count() == 0);
    assert(canvas.shapes.empty());

    std::cout << "Test test_Canvas_addNull: PASS" << std::endl;
}

// Test adding valid shapes and verifying count
void test_Canvas_addAndCount() {
    Canvas canvas;
    
    canvas.add(new Rectangle(2.0, 3.0));
    assert(canvas.count() == 1);

    canvas.add(new Circle(1.0));
    assert(canvas.count() == 2);

    canvas.add(new Shape());
    assert(canvas.count() == 3);

    std::cout << "Test test_Canvas_addAndCount: PASS" << std::endl;
}

// Test totalArea calculation
void test_Canvas_totalArea() {
    Canvas canvas;
    
    // Rectangle area: 4.0 * 5.0 = 20.0
    canvas.add(new Rectangle(4.0, 5.0));
    assert(isClose(canvas.totalArea(), 20.0));

    // Base Shape area: 0.0
    canvas.add(new Shape());
    assert(isClose(canvas.totalArea(), 20.0));

    // Circle area: PI * 2^2 = 4 * PI ≈ 12.566370614
    canvas.add(new Circle(2.0));
    double expectedArea = 20.0 + (3.14159265358979323846 * 4.0);
    assert(isClose(canvas.totalArea(), expectedArea));

    std::cout << "Test test_Canvas_totalArea: PASS" << std::endl;
}

// Test largestArea calculation
void test_Canvas_largestArea() {
    Canvas canvas;

    // Empty canvas largestArea should be 0.0
    assert(isClose(canvas.largestArea(), 0.0));

    // Add shapes with varying areas
    canvas.add(new Rectangle(2.0, 2.0));  // Area = 4.0
    assert(isClose(canvas.largestArea(), 4.0));

    canvas.add(new Rectangle(5.0, 5.0));  // Area = 25.0
    assert(isClose(canvas.largestArea(), 25.0));

    canvas.add(new Circle(1.0));          // Area ≈ 3.14159
    assert(isClose(canvas.largestArea(), 25.0));

    canvas.add(new Circle(3.0));          // Area ≈ 28.27433
    double expectedLargest = 3.14159265358979323846 * 9.0;
    assert(isClose(canvas.largestArea(), expectedLargest));

    std::cout << "Test test_Canvas_largestArea: PASS" << std::endl;
}

// Test clear() functionality
void test_Canvas_clear() {
    Canvas canvas;
    
    canvas.add(new Rectangle(10.0, 10.0));
    canvas.add(new Circle(5.0));
    assert(canvas.count() == 2);

    canvas.clear();
    assert(canvas.count() == 0);
    assert(canvas.shapes.empty());
    assert(isClose(canvas.totalArea(), 0.0));
    assert(isClose(canvas.largestArea(), 0.0));

    // Adding shapes after clear() to ensure Canvas is still functional
    canvas.add(new Rectangle(3.0, 4.0));
    assert(canvas.count() == 1);
    assert(isClose(canvas.totalArea(), 12.0));

    std::cout << "Test test_Canvas_clear: PASS" << std::endl;
}

// Test destructor cleanup behavior (verifies no leaks or crashes on destruction)
void test_Canvas_destructor() {
    {
        Canvas localCanvas;
        localCanvas.add(new Rectangle(2.0, 3.0));
        localCanvas.add(new Circle(4.0));
        localCanvas.add(new Shape());
    } // localCanvas goes out of scope here and its destructor should invoke clear() safely

    std::cout << "Test test_Canvas_destructor: PASS" << std::endl;
}

int main() {
    std::cout << "=== Running Canvas Unit Tests ===" << std::endl;

    test_Canvas_initialState();
    test_Canvas_addNull();
    test_Canvas_addAndCount();
    test_Canvas_totalArea();
    test_Canvas_largestArea();
    test_Canvas_clear();
    test_Canvas_destructor();

    std::cout << "=== All Canvas tests passed successfully! ===" << std::endl;
    return 0;
}
```
