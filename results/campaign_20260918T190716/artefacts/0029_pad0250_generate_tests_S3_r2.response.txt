Here is the complete C++ unit test suite for testing `Canvas.cpp` and its interactions with `Shape` objects.

```cpp
#include <iostream>
#include <cassert>
#include <cmath>
#include "Canvas.h"
#include "Shape.h"

// Helper function for floating point comparisons
bool isClose(double a, double b, double epsilon = 1e-6) {
    return std::fabs(a - b) < epsilon;
}

void test_Canvas_initialState() {
    Canvas canvas;
    assert(canvas.count() == 0);
    assert(isClose(canvas.totalArea(), 0.0));
    assert(isClose(canvas.largestArea(), 0.0));
    assert(canvas.shapes.empty());

    std::cout << "Test test_Canvas_initialState: PASS" << std::endl;
}

void test_Canvas_addNullShape() {
    Canvas canvas;
    canvas.add(nullptr);
    assert(canvas.count() == 0);
    assert(isClose(canvas.totalArea(), 0.0));
    assert(isClose(canvas.largestArea(), 0.0));

    std::cout << "Test test_Canvas_addNullShape: PASS" << std::endl;
}

void test_Canvas_addShapesAndCount() {
    Canvas canvas;
    canvas.add(new Rectangle(2.0, 3.0));
    assert(canvas.count() == 1);

    canvas.add(new Circle(1.0));
    assert(canvas.count() == 2);

    canvas.add(new Shape());
    assert(canvas.count() == 3);

    // Adding null should not increase count
    canvas.add(nullptr);
    assert(canvas.count() == 3);

    std::cout << "Test test_Canvas_addShapesAndCount: PASS" << std::endl;
}

void test_Canvas_totalArea() {
    Canvas canvas;
    
    // Rectangle 1: 4.0 * 5.0 = 20.0
    canvas.add(new Rectangle(4.0, 5.0));
    // Circle 1: PI * 2^2 = 4 * PI ≈ 12.566370614
    canvas.add(new Circle(2.0));
    // Generic shape: 0.0
    canvas.add(new Shape());

    double expectedTotal = 20.0 + (3.14159265358979323846 * 4.0) + 0.0;
    assert(isClose(canvas.totalArea(), expectedTotal));

    std::cout << "Test test_Canvas_totalArea: PASS" << std::endl;
}

void test_Canvas_largestArea() {
    Canvas canvas;

    // Empty canvas largest area is 0.0
    assert(isClose(canvas.largestArea(), 0.0));

    canvas.add(new Rectangle(2.0, 3.0)); // Area = 6.0
    assert(isClose(canvas.largestArea(), 6.0));

    canvas.add(new Circle(3.0)); // Area = PI * 9 ≈ 28.27433388
    double expectedCircleArea = 3.14159265358979323846 * 9.0;
    assert(isClose(canvas.largestArea(), expectedCircleArea));

    canvas.add(new Rectangle(1.0, 1.0)); // Area = 1.0 (smaller, largest should remain unchanged)
    assert(isClose(canvas.largestArea(), expectedCircleArea));

    canvas.add(new Rectangle(10.0, 10.0)); // Area = 100.0 (new largest)
    assert(isClose(canvas.largestArea(), 100.0));

    std::cout << "Test test_Canvas_largestArea: PASS" << std::endl;
}

void test_Canvas_clear() {
    Canvas canvas;
    canvas.add(new Rectangle(3.0, 4.0));
    canvas.add(new Circle(5.0));
    assert(canvas.count() == 2);

    canvas.clear();
    assert(canvas.count() == 0);
    assert(canvas.shapes.empty());
    assert(isClose(canvas.totalArea(), 0.0));
    assert(isClose(canvas.largestArea(), 0.0));

    // Calling clear again on an empty canvas should be safe
    canvas.clear();
    assert(canvas.count() == 0);

    // Adding shapes again after clear
    canvas.add(new Rectangle(2.0, 2.0));
    assert(canvas.count() == 1);
    assert(isClose(canvas.totalArea(), 4.0));

    std::cout << "Test test_Canvas_clear: PASS" << std::endl;
}

void test_Canvas_destructor() {
    // Tests that dynamically allocating and deallocating a Canvas with shapes runs cleanly
    Canvas* canvas = new Canvas();
    canvas->add(new Rectangle(2.0, 5.0));
    canvas->add(new Circle(1.5));
    delete canvas; // Invokes ~Canvas() which calls clear() and deletes shapes

    std::cout << "Test test_Canvas_destructor: PASS" << std::endl;
}

int main() {
    std::cout << "Starting Canvas test suite..." << std::endl;

    test_Canvas_initialState();
    test_Canvas_addNullShape();
    test_Canvas_addShapesAndCount();
    test_Canvas_totalArea();
    test_Canvas_largestArea();
    test_Canvas_clear();
    test_Canvas_destructor();

    std::cout << "All Canvas tests passed successfully!" << std::endl;
    return 0;
}
```
