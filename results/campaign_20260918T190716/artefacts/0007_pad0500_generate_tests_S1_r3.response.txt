Here is the complete unit test suite testing the `Canvas` class and its functionality.

```cpp
#include <iostream>
#include <cassert>
#include <cmath>
#include "Canvas.h"
#include "Shape.h"

// Helper function for comparing floating point values
static bool isClose(double a, double b, double epsilon = 1e-6) {
    return std::fabs(a - b) < epsilon;
}

void test_Canvas_initialState() {
    Canvas canvas;
    assert(canvas.count() == 0);
    assert(canvas.shapes.empty());
    assert(isClose(canvas.totalArea(), 0.0));
    assert(isClose(canvas.largestArea(), 0.0));
    std::cout << "Test test_Canvas_initialState: PASS" << std::endl;
}

void test_Canvas_addNullPointer() {
    Canvas canvas;
    canvas.add(nullptr);
    assert(canvas.count() == 0);
    assert(canvas.shapes.empty());
    assert(isClose(canvas.totalArea(), 0.0));
    assert(isClose(canvas.largestArea(), 0.0));
    std::cout << "Test test_Canvas_addNullPointer: PASS" << std::endl;
}

void test_Canvas_addAndCount() {
    Canvas canvas;
    
    Rectangle* rect = new Rectangle(4.0, 5.0);
    canvas.add(rect);
    assert(canvas.count() == 1);
    assert(canvas.shapes[0] == rect);

    Circle* circle = new Circle(3.0);
    canvas.add(circle);
    assert(canvas.count() == 2);
    assert(canvas.shapes[1] == circle);

    // Adding null should not increase count
    canvas.add(nullptr);
    assert(canvas.count() == 2);

    std::cout << "Test test_Canvas_addAndCount: PASS" << std::endl;
}

void test_Canvas_totalArea() {
    Canvas canvas;

    Rectangle* rect1 = new Rectangle(2.0, 3.0); // Area: 6.0
    Rectangle* rect2 = new Rectangle(4.0, 5.0); // Area: 20.0
    Circle* circle = new Circle(2.0);           // Area: PI * 4 ≈ 12.5663706...

    canvas.add(rect1);
    canvas.add(rect2);
    canvas.add(circle);

    double expectedArea = (2.0 * 3.0) + (4.0 * 5.0) + (3.14159265358979323846 * 4.0);
    assert(isClose(canvas.totalArea(), expectedArea));

    std::cout << "Test test_Canvas_totalArea: PASS" << std::endl;
}

void test_Canvas_largestArea() {
    Canvas canvas;

    Rectangle* smallRect = new Rectangle(2.0, 3.0); // Area: 6.0
    Rectangle* largeRect = new Rectangle(10.0, 5.0); // Area: 50.0
    Circle* mediumCircle = new Circle(3.0);         // Area: PI * 9 ≈ 28.27433388

    canvas.add(smallRect);
    assert(isClose(canvas.largestArea(), 6.0));

    canvas.add(largeRect);
    assert(isClose(canvas.largestArea(), 50.0));

    canvas.add(mediumCircle);
    // largeRect is still the largest
    assert(isClose(canvas.largestArea(), 50.0));

    // Add an even larger circle: radius = 5, area = PI * 25 ≈ 78.5398
    Circle* largestCircle = new Circle(5.0);
    canvas.add(largestCircle);
    assert(isClose(canvas.largestArea(), 3.14159265358979323846 * 25.0));

    std::cout << "Test test_Canvas_largestArea: PASS" << std::endl;
}

void test_Canvas_clear() {
    Canvas canvas;

    canvas.add(new Rectangle(2.0, 4.0));
    canvas.add(new Circle(1.5));
    canvas.add(new Shape());
    assert(canvas.count() == 3);

    canvas.clear();
    assert(canvas.count() == 0);
    assert(canvas.shapes.empty());
    assert(isClose(canvas.totalArea(), 0.0));
    assert(isClose(canvas.largestArea(), 0.0));

    // Can add elements again after clearing
    canvas.add(new Rectangle(3.0, 3.0));
    assert(canvas.count() == 1);
    assert(isClose(canvas.totalArea(), 9.0));
    assert(isClose(canvas.largestArea(), 9.0));

    std::cout << "Test test_Canvas_clear: PASS" << std::endl;
}

void test_Canvas_destructor() {
    // Tests that dynamic allocation and destruction of Canvas properly deletes shapes without leaks/crashes
    Canvas* canvas = new Canvas();
    canvas->add(new Rectangle(5.0, 5.0));
    canvas->add(new Circle(2.0));
    assert(canvas->count() == 2);
    
    delete canvas; // Calls destructor which in turn calls clear()

    std::cout << "Test test_Canvas_destructor: PASS" << std::endl;
}

int main() {
    std::cout << "Starting Canvas test suite..." << std::endl;

    test_Canvas_initialState();
    test_Canvas_addNullPointer();
    test_Canvas_addAndCount();
    test_Canvas_totalArea();
    test_Canvas_largestArea();
    test_Canvas_clear();
    test_Canvas_destructor();

    std::cout << "All Canvas tests passed successfully!" << std::endl;
    return 0;
}
```
