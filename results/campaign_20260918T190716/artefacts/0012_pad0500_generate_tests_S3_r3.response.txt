Here is the complete C++ unit test suite for testing `Canvas.cpp` and its interactions with `Shape` objects.

```cpp
#include <iostream>
#include <cassert>
#include <cmath>
#include "Canvas.h"
#include "Shape.h"

// Helper function for comparing floating-point numbers
bool isClose(double a, double b, double epsilon = 1e-6) {
    return std::fabs(a - b) < epsilon;
}

// Test initial state of a newly constructed Canvas
void test_Canvas_initialState() {
    Canvas canvas;
    assert(canvas.count() == 0);
    assert(isClose(canvas.totalArea(), 0.0));
    assert(isClose(canvas.largestArea(), 0.0));
    assert(canvas.shapes.empty());

    std::cout << "Test test_Canvas_initialState: PASS" << std::endl;
}

// Test adding null pointers to the Canvas
void test_Canvas_addNull() {
    Canvas canvas;
    canvas.add(nullptr);
    assert(canvas.count() == 0);
    assert(canvas.shapes.empty());
    assert(isClose(canvas.totalArea(), 0.0));
    assert(isClose(canvas.largestArea(), 0.0));

    std::cout << "Test test_Canvas_addNull: PASS" << std::endl;
}

// Test adding valid shapes to the Canvas and verify count
void test_Canvas_addValidShapes() {
    Canvas canvas;
    canvas.add(new Rectangle(4.0, 5.0));
    assert(canvas.count() == 1);

    canvas.add(new Circle(3.0));
    assert(canvas.count() == 2);

    canvas.add(new Shape());
    assert(canvas.count() == 3);

    // Ensure adding null afterwards still doesn't alter count
    canvas.add(nullptr);
    assert(canvas.count() == 3);

    std::cout << "Test test_Canvas_addValidShapes: PASS" << std::endl;
}

// Test calculating the total area of shapes in the Canvas
void test_Canvas_totalArea() {
    Canvas canvas;

    // Rectangle: 4.0 * 5.0 = 20.0
    canvas.add(new Rectangle(4.0, 5.0));
    assert(isClose(canvas.totalArea(), 20.0));

    // Circle: PI * 2.0^2 = 4.0 * PI
    const double PI = 3.14159265358979323846;
    canvas.add(new Circle(2.0));
    assert(isClose(canvas.totalArea(), 20.0 + 4.0 * PI));

    // Generic Shape has area 0.0
    canvas.add(new Shape());
    assert(isClose(canvas.totalArea(), 20.0 + 4.0 * PI));

    std::cout << "Test test_Canvas_totalArea: PASS" << std::endl;
}

// Test finding the largest area among shapes in the Canvas
void test_Canvas_largestArea() {
    Canvas canvas;

    // Empty canvas largest area should be 0.0
    assert(isClose(canvas.largestArea(), 0.0));

    // Add rectangle: area = 10.0 (2 * 5)
    canvas.add(new Rectangle(2.0, 5.0));
    assert(isClose(canvas.largestArea(), 10.0));

    // Add circle: area = PI * 3^2 ≈ 28.27433388
    const double PI = 3.14159265358979323846;
    canvas.add(new Circle(3.0));
    assert(isClose(canvas.largestArea(), 9.0 * PI));

    // Add smaller rectangle: area = 1.0 (1 * 1) -> largest should still be circle
    canvas.add(new Rectangle(1.0, 1.0));
    assert(isClose(canvas.largestArea(), 9.0 * PI));

    // Add larger rectangle: area = 100.0 (10 * 10)
    canvas.add(new Rectangle(10.0, 10.0));
    assert(isClose(canvas.largestArea(), 100.0));

    std::cout << "Test test_Canvas_largestArea: PASS" << std::endl;
}

// Test the clear() method to ensure elements are freed and container is reset
void test_Canvas_clear() {
    Canvas canvas;
    canvas.add(new Rectangle(3.0, 4.0));
    canvas.add(new Circle(5.0));
    canvas.add(new Shape());
    assert(canvas.count() == 3);

    canvas.clear();
    assert(canvas.count() == 0);
    assert(canvas.shapes.empty());
    assert(isClose(canvas.totalArea(), 0.0));
    assert(isClose(canvas.largestArea(), 0.0));

    // Ensure we can add new shapes cleanly after clear()
    canvas.add(new Rectangle(2.0, 3.0));
    assert(canvas.count() == 1);
    assert(isClose(canvas.totalArea(), 6.0));
    assert(isClose(canvas.largestArea(), 6.0));

    std::cout << "Test test_Canvas_clear: PASS" << std::endl;
}

// Test destructor execution and cleanup when Canvas goes out of scope
void test_Canvas_destructor() {
    // Canvas destructor calls clear() which deletes allocated shapes
    {
        Canvas canvas;
        canvas.add(new Rectangle(10.0, 20.0));
        canvas.add(new Circle(5.0));
    }
    // If memory management has issues or fails to compile/run, an error/leak occurs.
    std::cout << "Test test_Canvas_destructor: PASS" << std::endl;
}

int main() {
    std::cout << "--- Running Canvas Unit Tests ---" << std::endl;

    test_Canvas_initialState();
    test_Canvas_addNull();
    test_Canvas_addValidShapes();
    test_Canvas_totalArea();
    test_Canvas_largestArea();
    test_Canvas_clear();
    test_Canvas_destructor();

    std::cout << "--- All Canvas tests passed successfully! ---" << std::endl;
    return 0;
}
```
