Here is the complete C++ unit test suite testing the `Canvas` class and its functionality.

```cpp
#include <iostream>
#include <cassert>
#include <cmath>
#include "Canvas.h"
#include "Shape.h"

// Helper function for floating point comparisons
static bool areClose(double a, double b, double epsilon = 1e-6) {
    return std::fabs(a - b) < epsilon;
}

// Custom mock shape to test destructor invocation and virtual calls
class TrackedShape : public Shape {
public:
    double customArea;
    static int deleteCount;

    TrackedShape(double areaVal = 0.0) : customArea(areaVal) {}

    virtual ~TrackedShape() {
        deleteCount++;
    }

    virtual double area() const {
        return customArea;
    }
};

int TrackedShape::deleteCount = 0;

void test_Canvas_initialState() {
    Canvas canvas;
    assert(canvas.count() == 0);
    assert(canvas.shapes.empty());
    assert(canvas.totalArea() == 0.0);
    assert(canvas.largestArea() == 0.0);

    std::cout << "Test test_Canvas_initialState: PASS" << std::endl;
}

void test_Canvas_addValidShapes() {
    Canvas canvas;
    
    Rectangle* rect = new Rectangle(4.0, 5.0);
    Circle* circle = new Circle(3.0);

    canvas.add(rect);
    assert(canvas.count() == 1);
    assert(canvas.shapes.size() == 1);
    assert(canvas.shapes[0] == rect);

    canvas.add(circle);
    assert(canvas.count() == 2);
    assert(canvas.shapes.size() == 2);
    assert(canvas.shapes[1] == circle);

    std::cout << "Test test_Canvas_addValidShapes: PASS" << std::endl;
}

void test_Canvas_addNullptr() {
    Canvas canvas;
    
    canvas.add(nullptr);
    assert(canvas.count() == 0);
    assert(canvas.shapes.empty());

    // Add a valid shape, then try adding nullptr again
    canvas.add(new Rectangle(2.0, 3.0));
    assert(canvas.count() == 1);

    canvas.add(nullptr);
    assert(canvas.count() == 1);

    std::cout << "Test test_Canvas_addNullptr: PASS" << std::endl;
}

void test_Canvas_totalArea() {
    Canvas canvas;

    // Rectangle: 4 * 5 = 20
    canvas.add(new Rectangle(4.0, 5.0));
    // Circle: PI * 2^2 = 4 * PI ≈ 12.566370614
    canvas.add(new Circle(2.0));
    // Generic Shape with 0 area
    canvas.add(new Shape());

    double expectedArea = 20.0 + (3.14159265358979323846 * 4.0) + 0.0;
    assert(areClose(canvas.totalArea(), expectedArea));

    std::cout << "Test test_Canvas_totalArea: PASS" << std::endl;
}

void test_Canvas_largestArea() {
    Canvas canvas;

    // Initially 0.0 for empty canvas
    assert(canvas.largestArea() == 0.0);

    // Add smaller shape first
    canvas.add(new Rectangle(2.0, 3.0)); // Area = 6.0
    assert(areClose(canvas.largestArea(), 6.0));

    // Add larger shape
    canvas.add(new Rectangle(5.0, 5.0)); // Area = 25.0
    assert(areClose(canvas.largestArea(), 25.0));

    // Add shape with intermediate area
    canvas.add(new Circle(2.0)); // Area ≈ 12.566
    assert(areClose(canvas.largestArea(), 25.0));

    std::cout << "Test test_Canvas_largestArea: PASS" << std::endl;
}

void test_Canvas_clear() {
    TrackedShape::deleteCount = 0;
    Canvas canvas;

    canvas.add(new TrackedShape(10.0));
    canvas.add(new TrackedShape(20.0));
    canvas.add(new TrackedShape(30.0));

    assert(canvas.count() == 3);
    assert(TrackedShape::deleteCount == 0);

    canvas.clear();

    // Verify shapes are deallocated and vector is cleared
    assert(canvas.count() == 0);
    assert(canvas.shapes.empty());
    assert(TrackedShape::deleteCount == 3);
    assert(canvas.totalArea() == 0.0);
    assert(canvas.largestArea() == 0.0);

    std::cout << "Test test_Canvas_clear: PASS" << std::endl;
}

void test_Canvas_destructor() {
    TrackedShape::deleteCount = 0;
    {
        Canvas canvas;
        canvas.add(new TrackedShape(5.0));
        canvas.add(new TrackedShape(15.0));
        assert(canvas.count() == 2);
    }
    // Destructor should have called clear() and deleted all shapes
    assert(TrackedShape::deleteCount == 2);

    std::cout << "Test test_Canvas_destructor: PASS" << std::endl;
}

int main() {
    std::cout << "=== Running Canvas Unit Tests ===" << std::endl;

    test_Canvas_initialState();
    test_Canvas_addValidShapes();
    test_Canvas_addNullptr();
    test_Canvas_totalArea();
    test_Canvas_largestArea();
    test_Canvas_clear();
    test_Canvas_destructor();

    std::cout << "=== All Canvas Unit Tests Passed! ===" << std::endl;
    return 0;
}
```
