Here is the complete C++ unit test suite covering `Canvas.cpp` and its interactions with `Shape`.

```cpp
#include <iostream>
#include <cassert>
#include <cmath>
#include "Canvas.h"
#include "Shape.h"

// Helper function for floating point comparison
static bool isClose(double a, double b, double epsilon = 1e-6) {
    return std::fabs(a - b) < epsilon;
}

// Custom shape to test polymorphic behavior and proper destruction
class DestructionTrackerShape : public Shape {
public:
    int* destructionCounter;
    double shapeArea;

    DestructionTrackerShape(int* counter, double a = 10.0) 
        : destructionCounter(counter), shapeArea(a) {}

    virtual ~DestructionTrackerShape() {
        if (destructionCounter != nullptr) {
            (*destructionCounter)++;
        }
    }

    double area() const override {
        return shapeArea;
    }
};

void test_Canvas_initialState() {
    Canvas canvas;
    assert(canvas.count() == 0);
    assert(canvas.shapes.empty());
    assert(isClose(canvas.totalArea(), 0.0));
    assert(isClose(canvas.largestArea(), 0.0));
    std::cout << "Test test_Canvas_initialState: PASS" << std::endl;
}

void test_Canvas_addValidShapes() {
    Canvas canvas;
    Shape* r1 = new Rectangle(4.0, 5.0);
    Shape* c1 = new Circle(2.0);

    canvas.add(r1);
    assert(canvas.count() == 1);
    assert(canvas.shapes[0] == r1);

    canvas.add(c1);
    assert(canvas.count() == 2);
    assert(canvas.shapes[1] == c1);

    std::cout << "Test test_Canvas_addValidShapes: PASS" << std::endl;
}

void test_Canvas_addNullShape() {
    Canvas canvas;
    canvas.add(nullptr);
    assert(canvas.count() == 0);
    assert(canvas.shapes.empty());

    Shape* r1 = new Rectangle(2.0, 3.0);
    canvas.add(r1);
    assert(canvas.count() == 1);

    canvas.add(nullptr);
    assert(canvas.count() == 1);

    std::cout << "Test test_Canvas_addNullShape: PASS" << std::endl;
}

void test_Canvas_totalArea() {
    Canvas canvas;

    // Total area of empty canvas
    assert(isClose(canvas.totalArea(), 0.0));

    // Rectangle area: 3 * 4 = 12
    canvas.add(new Rectangle(3.0, 4.0));
    assert(isClose(canvas.totalArea(), 12.0));

    // Circle area: PI * 1^2 = 3.14159265358979323846
    canvas.add(new Circle(1.0));
    double expectedArea = 12.0 + (3.14159265358979323846 * 1.0 * 1.0);
    assert(isClose(canvas.totalArea(), expectedArea));

    // Generic shape area: 0
    canvas.add(new Shape());
    assert(isClose(canvas.totalArea(), expectedArea));

    std::cout << "Test test_Canvas_totalArea: PASS" << std::endl;
}

void test_Canvas_largestArea() {
    Canvas canvas;

    // Empty canvas largest area should be 0.0
    assert(isClose(canvas.largestArea(), 0.0));

    // Add first shape (Area: 10.0)
    canvas.add(new Rectangle(2.0, 5.0));
    assert(isClose(canvas.largestArea(), 10.0));

    // Add smaller shape (Area: 6.0)
    canvas.add(new Rectangle(2.0, 3.0));
    assert(isClose(canvas.largestArea(), 10.0));

    // Add larger shape (Circle radius 3 -> Area: ~28.2743)
    canvas.add(new Circle(3.0));
    double expectedLargest = 3.14159265358979323846 * 9.0;
    assert(isClose(canvas.largestArea(), expectedLargest));

    std::cout << "Test test_Canvas_largestArea: PASS" << std::endl;
}

void test_Canvas_clear() {
    Canvas canvas;
    int deleteCount = 0;

    canvas.add(new DestructionTrackerShape(&deleteCount));
    canvas.add(new DestructionTrackerShape(&deleteCount));
    canvas.add(new DestructionTrackerShape(&deleteCount));

    assert(canvas.count() == 3);
    assert(deleteCount == 0);

    canvas.clear();

    assert(canvas.count() == 0);
    assert(canvas.shapes.empty());
    assert(deleteCount == 3);
    assert(isClose(canvas.totalArea(), 0.0));
    assert(isClose(canvas.largestArea(), 0.0));

    // Calling clear again on empty canvas should be safe
    canvas.clear();
    assert(canvas.count() == 0);
    assert(deleteCount == 3);

    std::cout << "Test test_Canvas_clear: PASS" << std::endl;
}

void test_Canvas_destructorCleansMemory() {
    int deleteCount = 0;
    {
        Canvas canvas;
        canvas.add(new DestructionTrackerShape(&deleteCount));
        canvas.add(new DestructionTrackerShape(&deleteCount));
        assert(canvas.count() == 2);
    }
    // Destructor should have been invoked when exiting scope
    assert(deleteCount == 2);

    std::cout << "Test test_Canvas_destructorCleansMemory: PASS" << std::endl;
}

int main() {
    std::cout << "=== Running Canvas Unit Tests ===" << std::endl;

    test_Canvas_initialState();
    test_Canvas_addValidShapes();
    test_Canvas_addNullShape();
    test_Canvas_totalArea();
    test_Canvas_largestArea();
    test_Canvas_clear();
    test_Canvas_destructorCleansMemory();

    std::cout << "=== All Canvas tests passed successfully! ===" << std::endl;
    return 0;
}
```
