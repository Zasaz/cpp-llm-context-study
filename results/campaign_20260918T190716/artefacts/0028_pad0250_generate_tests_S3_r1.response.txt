Here is the complete C++ unit test suite for testing the `Canvas` class and its methods.

```cpp
#include <iostream>
#include <cassert>
#include <cmath>
#include "Canvas.h"
#include "Shape.h"

// Helper function for floating point comparison
bool areAlmostEqual(double a, double b, double epsilon = 1e-6) {
    return std::fabs(a - b) < epsilon;
}

// Test initial state of a newly constructed Canvas
void test_Canvas_initialState() {
    Canvas canvas;
    assert(canvas.count() == 0);
    assert(areAlmostEqual(canvas.totalArea(), 0.0));
    assert(areAlmostEqual(canvas.largestArea(), 0.0));
    assert(canvas.shapes.empty());

    std::cout << "Test test_Canvas_initialState: PASS" << std::endl;
}

// Test adding valid shapes and null pointers to the Canvas
void test_Canvas_add() {
    Canvas canvas;

    // Adding null pointer should be ignored
    canvas.add(nullptr);
    assert(canvas.count() == 0);

    // Adding valid shapes
    canvas.add(new Rectangle(2.0, 3.0));
    assert(canvas.count() == 1);

    canvas.add(new Circle(1.0));
    assert(canvas.count() == 2);

    // Adding another null pointer should not change count
    canvas.add(nullptr);
    assert(canvas.count() == 2);

    std::cout << "Test test_Canvas_add: PASS" << std::endl;
}

// Test totalArea calculation with multiple shapes
void test_Canvas_totalArea() {
    Canvas canvas;
    assert(areAlmostEqual(canvas.totalArea(), 0.0));

    // Rectangle: 4.0 * 5.0 = 20.0
    canvas.add(new Rectangle(4.0, 5.0));
    assert(areAlmostEqual(canvas.totalArea(), 20.0));

    // Circle: PI * 2.0^2 = 4 * PI (~12.5663706)
    canvas.add(new Circle(2.0));
    double expectedArea = 20.0 + (3.14159265358979323846 * 4.0);
    assert(areAlmostEqual(canvas.totalArea(), expectedArea));

    // Base Shape has area 0.0
    canvas.add(new Shape());
    assert(areAlmostEqual(canvas.totalArea(), expectedArea));

    std::cout << "Test test_Canvas_totalArea: PASS" << std::endl;
}

// Test largestArea calculation across different shapes and orderings
void test_Canvas_largestArea() {
    Canvas canvas;
    assert(areAlmostEqual(canvas.largestArea(), 0.0));

    // Add smaller shape first: Circle area = PI * 1^2 ~= 3.14159
    canvas.add(new Circle(1.0));
    assert(areAlmostEqual(canvas.largestArea(), 3.14159265358979323846));

    // Add larger shape: Rectangle area = 5.0 * 2.0 = 10.0
    canvas.add(new Rectangle(5.0, 2.0));
    assert(areAlmostEqual(canvas.largestArea(), 10.0));

    // Add smaller shape: Rectangle area = 2.0 * 2.0 = 4.0
    canvas.add(new Rectangle(2.0, 2.0));
    assert(areAlmostEqual(canvas.largestArea(), 10.0));

    // Add largest shape: Circle area = PI * 3^2 ~= 28.27433
    canvas.add(new Circle(3.0));
    double expectedLargest = 3.14159265358979323846 * 9.0;
    assert(areAlmostEqual(canvas.largestArea(), expectedLargest));

    std::cout << "Test test_Canvas_largestArea: PASS" << std::endl;
}

// Test clear functionality
void test_Canvas_clear() {
    Canvas canvas;

    // Clear empty canvas should be safe
    canvas.clear();
    assert(canvas.count() == 0);

    canvas.add(new Rectangle(3.0, 4.0));
    canvas.add(new Circle(2.0));
    assert(canvas.count() == 2);

    canvas.clear();
    assert(canvas.count() == 0);
    assert(areAlmostEqual(canvas.totalArea(), 0.0));
    assert(areAlmostEqual(canvas.largestArea(), 0.0));
    assert(canvas.shapes.empty());

    // Should be able to add new shapes after clearing
    canvas.add(new Rectangle(1.0, 1.0));
    assert(canvas.count() == 1);
    assert(areAlmostEqual(canvas.totalArea(), 1.0));

    std::cout << "Test test_Canvas_clear: PASS" << std::endl;
}

// Test destructor cleanup (ensures destructor executes properly via scope)
void test_Canvas_destructor() {
    {
        Canvas canvas;
        canvas.add(new Rectangle(10.0, 10.0));
        canvas.add(new Circle(5.0));
        canvas.add(new Shape());
        // canvas goes out of scope here; its destructor calls clear() and deletes shapes
    }

    std::cout << "Test test_Canvas_destructor: PASS" << std::endl;
}

int main() {
    std::cout << "=== Running Canvas Unit Tests ===" << std::endl;

    test_Canvas_initialState();
    test_Canvas_add();
    test_Canvas_totalArea();
    test_Canvas_largestArea();
    test_Canvas_clear();
    test_Canvas_destructor();

    std::cout << "=== All Canvas tests passed successfully! ===" << std::endl;
    return 0;
}
```
