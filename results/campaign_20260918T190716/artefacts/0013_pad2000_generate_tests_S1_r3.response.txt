Here is the complete C++ unit test suite covering the `Canvas` class and its functionality.

```cpp
#include <iostream>
#include <cassert>
#include <cmath>
#include "Canvas.h"
#include "Shape.h"

// Helper function for floating point comparisons
static bool isClose(double a, double b, double epsilon = 1e-6) {
    return std::fabs(a - b) < epsilon;
}

void test_Canvas_initialState() {
    Canvas canvas;
    assert(canvas.count() == 0);
    assert(canvas.shapes.empty());
    assert(isClose(canvas.totalArea(), 0.0));
    assert(isClose(canvas.largestArea(), 0.0));
    std::cout << "Test test_Canvas_initialState: PASS" << std::endl;
}

void test_Canvas_addValidShapes() {
    Canvas canvas;
    Rectangle* rect = new Rectangle(4.0, 5.0); // Area: 20.0
    Circle* circle = new Circle(2.0);          // Area: ~12.56637

    canvas.add(rect);
    assert(canvas.count() == 1);
    assert(canvas.shapes[0] == rect);

    canvas.add(circle);
    assert(canvas.count() == 2);
    assert(canvas.shapes[1] == circle);

    std::cout << "Test test_Canvas_addValidShapes: PASS" << std::endl;
}

void test_Canvas_addNullShape() {
    Canvas canvas;
    canvas.add(nullptr);
    assert(canvas.count() == 0);
    assert(canvas.shapes.empty());

    Rectangle* rect = new Rectangle(3.0, 3.0);
    canvas.add(rect);
    assert(canvas.count() == 1);

    canvas.add(nullptr);
    assert(canvas.count() == 1);

    std::cout << "Test test_Canvas_addNullShape: PASS" << std::endl;
}

void test_Canvas_totalArea() {
    Canvas canvas;
    assert(isClose(canvas.totalArea(), 0.0));

    Rectangle* rect1 = new Rectangle(2.0, 3.0); // Area = 6.0
    Rectangle* rect2 = new Rectangle(4.0, 5.0); // Area = 20.0
    Circle* circ = new Circle(1.0);             // Area = PI * 1^2 ≈ 3.141592653589793

    canvas.add(rect1);
    assert(isClose(canvas.totalArea(), 6.0));

    canvas.add(rect2);
    assert(isClose(canvas.totalArea(), 26.0));

    canvas.add(circ);
    double expectedTotal = 6.0 + 20.0 + 3.14159265358979323846;
    assert(isClose(canvas.totalArea(), expectedTotal));

    std::cout << "Test test_Canvas_totalArea: PASS" << std::endl;
}

void test_Canvas_largestArea() {
    Canvas canvas;
    assert(isClose(canvas.largestArea(), 0.0));

    Rectangle* rectSmall = new Rectangle(2.0, 2.0); // Area = 4.0
    Rectangle* rectLarge = new Rectangle(10.0, 5.0); // Area = 50.0
    Circle* circMedium = new Circle(3.0);            // Area ≈ 28.27433

    canvas.add(rectSmall);
    assert(isClose(canvas.largestArea(), 4.0));

    canvas.add(rectLarge);
    assert(isClose(canvas.largestArea(), 50.0));

    canvas.add(circMedium);
    assert(isClose(canvas.largestArea(), 50.0));

    std::cout << "Test test_Canvas_largestArea: PASS" << std::endl;
}

void test_Canvas_clear() {
    Canvas canvas;
    canvas.add(new Rectangle(2.0, 5.0));
    canvas.add(new Circle(4.0));
    canvas.add(new Shape());
    assert(canvas.count() == 3);

    canvas.clear();
    assert(canvas.count() == 0);
    assert(canvas.shapes.empty());
    assert(isClose(canvas.totalArea(), 0.0));
    assert(isClose(canvas.largestArea(), 0.0));

    // Clearing an already empty canvas should be safe
    canvas.clear();
    assert(canvas.count() == 0);

    // Can add shapes again after clear
    canvas.add(new Rectangle(3.0, 3.0));
    assert(canvas.count() == 1);
    assert(isClose(canvas.totalArea(), 9.0));

    std::cout << "Test test_Canvas_clear: PASS" << std::endl;
}

void test_Canvas_destructor() {
    // Tests that dynamically allocated Canvas cleans up without memory/runtime errors
    Canvas* canvas = new Canvas();
    canvas->add(new Rectangle(2.0, 3.0));
    canvas->add(new Circle(1.5));
    delete canvas;

    std::cout << "Test test_Canvas_destructor: PASS" << std::endl;
}

int main() {
    test_Canvas_initialState();
    test_Canvas_addValidShapes();
    test_Canvas_addNullShape();
    test_Canvas_totalArea();
    test_Canvas_largestArea();
    test_Canvas_clear();
    test_Canvas_destructor();

    std::cout << "\nAll Canvas tests passed successfully!" << std::endl;
    return 0;
}
```
