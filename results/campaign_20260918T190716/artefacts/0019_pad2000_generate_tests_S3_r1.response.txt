Here is the complete C++ unit test suite for testing `Canvas.cpp`.

```cpp
#include <iostream>
#include <cassert>
#include <cmath>
#include "Canvas.h"
#include "Shape.h"

// Helper function for floating point comparisons
static bool approxEqual(double a, double b, double epsilon = 1e-6) {
    return std::fabs(a - b) < epsilon;
}

// Test initial/empty state of Canvas
void test_Canvas_initialState() {
    Canvas canvas;
    assert(canvas.count() == 0);
    assert(canvas.shapes.empty());
    assert(approxEqual(canvas.totalArea(), 0.0));
    assert(approxEqual(canvas.largestArea(), 0.0));

    std::cout << "Test test_Canvas_initialState: PASS" << std::endl;
}

// Test adding null pointer to Canvas
void test_Canvas_addNull() {
    Canvas canvas;
    canvas.add(nullptr);
    assert(canvas.count() == 0);
    assert(canvas.shapes.empty());
    assert(approxEqual(canvas.totalArea(), 0.0));
    assert(approxEqual(canvas.largestArea(), 0.0));

    std::cout << "Test test_Canvas_addNull: PASS" << std::endl;
}

// Test adding valid shapes to Canvas and count()
void test_Canvas_addAndCount() {
    Canvas canvas;
    canvas.add(new Rectangle(4.0, 5.0));
    assert(canvas.count() == 1);

    canvas.add(new Circle(3.0));
    assert(canvas.count() == 2);

    canvas.add(nullptr); // should not be added
    assert(canvas.count() == 2);

    std::cout << "Test test_Canvas_addAndCount: PASS" << std::endl;
}

// Test totalArea() calculation
void test_Canvas_totalArea() {
    Canvas canvas;
    // Rectangle area: 4.0 * 5.0 = 20.0
    canvas.add(new Rectangle(4.0, 5.0));
    assert(approxEqual(canvas.totalArea(), 20.0));

    // Circle area: PI * 2.0 * 2.0 ≈ 12.566370614
    canvas.add(new Circle(2.0));
    double expectedTotal = 20.0 + (3.14159265358979323846 * 4.0);
    assert(approxEqual(canvas.totalArea(), expectedTotal));

    // Base Shape area: 0.0
    canvas.add(new Shape());
    assert(approxEqual(canvas.totalArea(), expectedTotal));

    std::cout << "Test test_Canvas_totalArea: PASS" << std::endl;
}

// Test largestArea() calculation
void test_Canvas_largestArea() {
    Canvas canvas;
    assert(approxEqual(canvas.largestArea(), 0.0));

    // Add rectangle: area = 10.0
    canvas.add(new Rectangle(2.0, 5.0));
    assert(approxEqual(canvas.largestArea(), 10.0));

    // Add circle: area = PI * 3^2 ≈ 28.2743338823 > 10.0
    canvas.add(new Circle(3.0));
    double circleArea = 3.14159265358979323846 * 9.0;
    assert(approxEqual(canvas.largestArea(), circleArea));

    // Add smaller rectangle: area = 6.0
    canvas.add(new Rectangle(2.0, 3.0));
    assert(approxEqual(canvas.largestArea(), circleArea));

    // Add larger rectangle: area = 50.0 > circleArea
    canvas.add(new Rectangle(5.0, 10.0));
    assert(approxEqual(canvas.largestArea(), 50.0));

    std::cout << "Test test_Canvas_largestArea: PASS" << std::endl;
}

// Test clear() functionality
void test_Canvas_clear() {
    Canvas canvas;
    canvas.add(new Rectangle(10.0, 2.0));
    canvas.add(new Circle(1.0));
    assert(canvas.count() == 2);

    canvas.clear();
    assert(canvas.count() == 0);
    assert(canvas.shapes.empty());
    assert(approxEqual(canvas.totalArea(), 0.0));
    assert(approxEqual(canvas.largestArea(), 0.0));

    // Ensure we can add again after clearing
    canvas.add(new Rectangle(3.0, 3.0));
    assert(canvas.count() == 1);
    assert(approxEqual(canvas.totalArea(), 9.0));
    assert(approxEqual(canvas.largestArea(), 9.0));

    std::cout << "Test test_Canvas_clear: PASS" << std::endl;
}

// Test Canvas destructor cleans up shapes safely
void test_Canvas_destructor() {
    // Canvas creates shapes on the heap, and its destructor calls clear() which deletes them.
    {
        Canvas canvas;
        canvas.add(new Rectangle(10.0, 10.0));
        canvas.add(new Circle(5.0));
        canvas.add(new Shape());
    } // canvas goes out of scope here; destructor executes without memory leak / crash.

    std::cout << "Test test_Canvas_destructor: PASS" << std::endl;
}

int main() {
    std::cout << "--- Starting Canvas Unit Tests ---" << std::endl;

    test_Canvas_initialState();
    test_Canvas_addNull();
    test_Canvas_addAndCount();
    test_Canvas_totalArea();
    test_Canvas_largestArea();
    test_Canvas_clear();
    test_Canvas_destructor();

    std::cout << "--- All Canvas tests PASSED successfully! ---" << std::endl;
    return 0;
}
```
